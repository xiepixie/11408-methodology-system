\node[below left=1pt] at (0,0) {$O$};

% 关键极值点的竖向参考线
\draw[dashed,line width=0.6pt] (1,-0.2) -- (1,0.9);

\node[below] at (1,0) {$1$};
\node[below] at (3,0) {$3$};

% 精确函数曲线

  plot (\x,{0.5*((\x*\x*\x)/3-2*\x*\x+3*\x)});
