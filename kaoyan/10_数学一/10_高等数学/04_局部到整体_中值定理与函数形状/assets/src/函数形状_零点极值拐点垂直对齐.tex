\documentclass[dvisvgm,tikz,border=3pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\usepackage{CJKutf8}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta,calc,positioning}

\definecolor{themebg}{HTML}{2e362c}
\definecolor{themefg}{HTML}{edf4e8}
\definecolor{themegray}{HTML}{9ea897}
\definecolor{themecurve}{HTML}{7eb6ff}
\definecolor{themealert}{HTML}{ff7b7b}
\definecolor{themeamber}{HTML}{f5b942}

\pgfdeclarelayer{background}
\pgfdeclarelayer{foreground}
\pgfsetlayers{background,main,foreground}

\begin{document}
\begin{CJK*}{UTF8}{gbsn}
\pagecolor{themebg}
\begin{tikzpicture}[color=themefg, text=themefg]

% ── 顶层: f(x) = x^3/3 - x ──
\begin{axis}[
    name=ax1,
    width=13.2cm,
    height=4.8cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\scriptsize},
    every axis label/.style={color=themefg},
    xmin=-2.5, xmax=2.5,
    ymin=-1.4, ymax=1.4,
    xtick={-1.732,-1,0,1,1.732},
    xticklabels={$-\sqrt{3}$,$-1$,$0$,$1$,$\sqrt{3}$},
    ytick=\empty,
    clip=false,
    title style={font=\footnotesize\bfseries, at={(0.5,1.15)}, anchor=south},
    title={函数形状三层共轴对齐：原函数 $f(x)$、导函数 $f'(x)$ 与二阶导 $f''(x)$},
]
\addplot[themecurve, line width=2.0pt, domain=-2.2:2.2, samples=100] {(x^3)/3 - x};
\node[text=themecurve, font=\footnotesize\bfseries, fill=themebg, inner sep=1pt, anchor=south west] at (axis cs:1.8,0.7) {$y=f(x)$};

\begin{pgfonlayer}{foreground}
    % 极大值 (-1, 2/3), 极小值 (1, -2/3), 拐点 (0,0)
    \addplot[only marks, mark=*, mark size=2.5pt, fill=themeamber, draw=themefg] coordinates {(-1,0.667) (1,-0.667)};
    \addplot[only marks, mark=*, mark size=2.5pt, fill=themealert, draw=themefg] coordinates {(0,0)};
    \node[text=themeamber, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:-1,0.75) {极大值};
    \node[text=themeamber, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=north] at (axis cs:1,-0.75) {极小值};
    \node[text=themealert, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south east] at (axis cs:-0.15,0.1) {拐点};

    \coordinate (p1_neg1) at (axis cs:-1, 1.35);
    \coordinate (p1_zero) at (axis cs:0, 1.35);
    \coordinate (p1_pos1) at (axis cs:1, 1.35);
\end{pgfonlayer}
\end{axis}

% ── 中层: f'(x) = x^2 - 1 ──
\begin{axis}[
    name=ax2,
    at={($(ax1.south)-(0,1.2cm)$)},
    anchor=north,
    width=13.2cm,
    height=4.4cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\scriptsize},
    every axis label/.style={color=themefg},
    xmin=-2.5, xmax=2.5,
    ymin=-1.4, ymax=1.8,
    xtick={-1,1},
    ytick=\empty,
    clip=false,
]
\addplot[themeamber, line width=1.8pt, domain=-2.0:2.0, samples=80] {x^2 - 1};
\node[text=themeamber, font=\footnotesize\bfseries, fill=themebg, inner sep=1pt, anchor=south west] at (axis cs:1.8,0.9) {$y=f'(x)$};

\begin{pgfonlayer}{foreground}
    % 零点 x = ±1
    \addplot[only marks, mark=*, mark size=2.5pt, fill=themeamber, draw=themefg] coordinates {(-1,0) (1,0)};
    \node[text=themeamber, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south east] at (axis cs:-1.1,0.15) {$f'=0$};
    \node[text=themeamber, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south west] at (axis cs:1.1,0.15) {$f'=0$};

    \node[text=themefg, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=north] at (axis cs:0,-1.05) {$f'<0$ (单减)};
    \node[text=themefg, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:-1.8,0.5) {$f'>0$ (单增)};
    \node[text=themefg, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:1.8,0.5) {$f'>0$ (单增)};
\end{pgfonlayer}
\end{axis}

% ── 底层: f''(x) = 2x ──
\begin{axis}[
    name=ax3,
    at={($(ax2.south)-(0,1.2cm)$)},
    anchor=north,
    width=13.2cm,
    height=4.4cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\scriptsize},
    every axis label/.style={color=themefg},
    xmin=-2.5, xmax=2.5,
    ymin=-2.2, ymax=2.2,
    xtick={-1,1},
    ytick=\empty,
    clip=false,
    xlabel={$x$},
]
\addplot[themealert, line width=1.8pt, domain=-1.8:1.8, samples=50] {2*x};
\node[text=themealert, font=\footnotesize\bfseries, fill=themebg, inner sep=1pt, anchor=south west] at (axis cs:1.4,1.4) {$y=f''(x)$};

\begin{pgfonlayer}{foreground}
    % 零点 x = 0
    \addplot[only marks, mark=*, mark size=2.5pt, fill=themealert, draw=themefg] coordinates {(0,0)};
    \node[text=themealert, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=north east] at (axis cs:-0.1,-0.1) {$f''=0$};

    \node[text=themealert, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:-1.4,-1.5) {$f''<0$ (凹下 / 凸)};
    \node[text=themealert, font=\scriptsize, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:1.4,0.6) {$f''>0$ (凹上 / 凹)};

    \coordinate (p3_neg1) at (axis cs:-1, -2.1);
    \coordinate (p3_zero) at (axis cs:0, -2.1);
    \coordinate (p3_pos1) at (axis cs:1, -2.1);
\end{pgfonlayer}
\end{axis}

% ── 跨层竖直虚线对齐 ──
\begin{pgfonlayer}{background}
    \draw[themeamber!70!themebg, dashed, line width=1pt] (p1_neg1) -- (p3_neg1);
    \draw[themealert!70!themebg, dashed, line width=1pt] (p1_zero) -- (p3_zero);
    \draw[themeamber!70!themebg, dashed, line width=1pt] (p1_pos1) -- (p3_pos1);
\end{pgfonlayer}

\end{tikzpicture}
\end{CJK*}
