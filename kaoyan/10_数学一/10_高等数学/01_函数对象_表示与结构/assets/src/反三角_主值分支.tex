    xtick={-3.14159, -1.5708, 0, 0.5236, 1.5708, 2.618, 3.14159},
    xticklabels={$-\pi$, $-\frac{\pi}{2}$, $0$, $\frac{\pi}{6}$, $\frac{\pi}{2}$, $\frac{5\pi}{6}$, $\pi$},
    ytick={-1, 0, 0.5, 1},

    clip=false,
    xlabel={$x$},

    title style={at={(0.5,1.08)},anchor=south,color=themefg,font=\bfseries\normalsize},

% full periodic sine curve (dashed)
\addplot[themegray, dashed, thick, domain=-4.2:4.2, samples=300] {sin(deg(x))};
\node[text=themegray, fill=themebg, inner sep=0.8pt, anchor=south west] at (axis cs:2.8,1.1) {\small $y=\sin x$（在 $\mathbb R$ 上非单射）};

% principal branch [-pi/2, pi/2]
\addplot[themecurve, line width=2.4pt, domain=-1.5708:1.5708, samples=200] {sin(deg(x))};

% boundary vertical dashed lines
\draw[dashed, themegray, thick] (axis cs:-1.5708,-1.4) -- (axis cs:-1.5708,1.4);
\draw[dashed, themegray, thick] (axis cs:1.5708,-1.4) -- (axis cs:1.5708,1.4);

% Endpoints of principal branch: (-pi/2, -1) and (pi/2, 1)
\addplot[only marks, mark=*, mark size=2.8pt, fill=themecurve, draw=themefg] coordinates {(-1.5708, -1) (1.5708, 1)};
\node[text=themecurve, fill=themebg, inner sep=0.8pt, below left] at (axis cs:-1.5708, -1) {\small $(-\frac{\pi}{2},-1)$};
\node[text=themecurve, fill=themebg, inner sep=0.8pt, above right] at (axis cs:1.5708, 1) {\small $(\frac{\pi}{2},1)$};

% horizontal test line y = 0.5
\draw[dashed, themegray, line width=1.1pt] (axis cs:-4.4, 0.5) -- (axis cs:4.4, 0.5);
\node[text=themegray, fill=themebg, inner sep=0.8pt, anchor=south east] at (axis cs:-3.9, 0.52) {\small $y=\frac{1}{2}$};

% principal intersection (pi/6, 0.5)
\addplot[only marks, mark=*, mark size=3.2pt, fill=themecurve, draw=themefg] coordinates {(0.5236, 0.5)};
\draw[dotted, themegray, thick] (axis cs:0.5236, 0.5) -- (axis cs:0.5236, 0);
\node[text=themecurve, fill=themebg, inner sep=0.8pt, above] at (axis cs:0.5236, 0.75) {\small 主值唯一原像 $\arcsin\frac{1}{2}=\frac{\pi}{6}$};

% other intersections on sine wave (outside principal branch)
\addplot[only marks, mark=o, mark size=3.2pt, thick, themegray] coordinates {(-3.6652, 0.5) (2.618, 0.5)};
\node[text=themegray, fill=themebg, inner sep=0.8pt, above] at (axis cs:2.618, 0.68) {\small 舍弃分支 $x=\frac{5\pi}{6}$};

\end{axis}
