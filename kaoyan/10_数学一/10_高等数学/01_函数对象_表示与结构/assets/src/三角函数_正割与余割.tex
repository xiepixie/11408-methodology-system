\begin{axis}[
    name=axleft,

    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\footnotesize},
    every axis label/.style={color=themefg},
    xmin=-4.2, xmax=4.2,
    ymin=-3.5, ymax=3.5,
    xtick={-3.14159, -1.5708, 0, 1.5708, 3.14159},
    xticklabels={$-\pi$, $-\frac{\pi}{2}$, $0$, $\frac{\pi}{2}$, $\pi$},
    ytick={-1, 0, 1},
    clip=true,
    xlabel={$x$},
    ylabel={$y$},
    title style={at={(0.5,1.05)},anchor=south,color=themefg,font=\bfseries\small},
    title={$y=\sec x=\frac{1}{\cos x}$（偶函数，值域 $\lvert y\rvert\ge1$）},
]
  % Vertical asymptotes
  \draw[dashed, themegray, thick] (axis cs:-1.5708,-3.5) -- (axis cs:-1.5708,3.5);
  \draw[dashed, themegray, thick] (axis cs:1.5708,-3.5) -- (axis cs:1.5708,3.5);

  \addplot[themecurve, line width=1.8pt, domain=-1.25:1.25, samples=150] {1/cos(deg(x))};
  % Lower branch around pi and -pi
  \addplot[themecurve, line width=1.8pt, domain=1.85:4.1, samples=150] {1/cos(deg(x))};
  \addplot[themecurve, line width=1.8pt, domain=-4.1:-1.85, samples=150] {1/cos(deg(x))};
  
  % Extrema points
  \addplot[only marks, mark=*, mark size=2.5pt, fill=themecurve, draw=themefg] coordinates {(0,1) (3.14159,-1) (-3.14159,-1)};
  \node[text=themecurve, fill=themebg, inner sep=0.8pt, above right] at (axis cs:0.1,1.05) {\footnotesize $(0,1)$};
  \node[text=themecurve, fill=themebg, inner sep=0.8pt, below] at (axis cs:3.14159,-1.1) {\footnotesize $(\pi,-1)$};
\end{axis}
