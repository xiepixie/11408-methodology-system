\begin{CJK*}{UTF8}{gbsn}

\begin{tikzpicture}[color=themefg, text=themefg]

% Left: y = cos x
\begin{axis}[
    name=axleft,
    width=7.2cm,
    height=6.2cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\footnotesize},
    every axis label/.style={color=themefg},
    xmin=-3.8, xmax=4.0,
    ymin=-1.5, ymax=1.6,
    xtick={-3.14159, -1.5708, 0, 1.5708, 3.14159},
    xticklabels={$-\pi$, $-\frac{\pi}{2}$, $0$, $\frac{\pi}{2}$, $\pi$},
    ytick={-1, 0, 1},
    clip=false,
    xlabel={$x$},
    ylabel={$y$},
    title style={at={(0.5,1.05)},anchor=south,color=themefg,font=\bfseries\small},
    title={$y=\cos x$（主值单射分支 $[0,\pi]$）},
]
  % full periodic cos
  \addplot[themegray, dashed, thick, domain=-3.5:3.5, samples=200] {cos(deg(x))};
  
  % principal branch [0, pi]
  \addplot[themecurve, line width=2pt, domain=0:3.14159, samples=200] {cos(deg(x))};
  
  % endpoints
  \addplot[only marks, mark=*, mark size=2.5pt, fill=themecurve, draw=themefg] coordinates {(0,1) (3.14159,-1)};
  \node[text=themecurve, fill=themebg, inner sep=0.8pt, above right] at (axis cs:0.1,1.05) {\footnotesize $(0,1)$};
  \node[text=themecurve, fill=themebg, inner sep=0.8pt, below left] at (axis cs:3.14159,-1.05) {\footnotesize $(\pi,-1)$};
\end{axis}

% Right: y = arccos x
\begin{axis}[
    at={($(axleft.east)+(2.2cm,0)$)},
    anchor=west,
    width=7.2cm,
    height=6.2cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\footnotesize},
    every axis label/.style={color=themefg},
    xmin=-1.6, xmax=1.8,
    ymin=-0.5, ymax=3.6,
    xtick={-1, 0, 1},
    ytick={0, 3.14159},
    yticklabels={$0$, $\pi$},
    clip=false,
    xlabel={$x$},
    ylabel={$y$},
    title style={at={(0.5,1.05)},anchor=south,color=themefg,font=\bfseries\small},
    title={$y=\arccos x$（定义域 $[-1,1]$，值域 $[0,\pi]$）},
]
  % arccos curve
  \addplot[themealert, line width=2pt, domain=-1:1, samples=200] {rad(acos(x))};
  
  % endpoints & midpoint
  \addplot[only marks, mark=*, mark size=2.5pt, fill=themealert, draw=themefg] coordinates {(-1, 3.14159) (0, 1.5708) (1, 0)};
  \node[text=themealert, fill=themebg, inner sep=0.8pt, above right] at (axis cs:-1, 3.14159) {\footnotesize $(-1,\pi)$};
  \node[text=themefg, fill=themebg, inner sep=0.8pt, above right] at (axis cs:0.1, 1.5708) {\footnotesize $(0,\frac{\pi}{2})$ 对称中心};
  \node[text=themealert, fill=themebg, inner sep=0.8pt, above right] at (axis cs:1, 0.15) {\footnotesize $(1,0)$};
  
  % dashed limit bounds
  \draw[dashed, themegray, thick] (axis cs:-1,0) -- (axis cs:-1,3.14159) -- (axis cs:0,3.14159);
\end{axis}

\end{tikzpicture}
