\documentclass[dvisvgm,tikz,border=3pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\usepackage{CJKutf8}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta,calc}

\definecolor{themebg}{HTML}{30362d}
\definecolor{themefg}{HTML}{edf4e8}
\definecolor{themegray}{HTML}{9ea897}
\definecolor{themecurve}{HTML}{7eb6ff}
\definecolor{themealert}{HTML}{ff7b7b}

\begin{document}
\begin{CJK*}{UTF8}{gbsn}
\pagecolor{themebg}
\begin{tikzpicture}[color=themefg, text=themefg]

% Left: Strictly Increasing (Injective)
\begin{axis}[
    name=axleft,
    width=7.5cm,
    height=7.2cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\small},
    every axis label/.style={color=themefg},
    xmin=-0.5, xmax=3.6,
    ymin=-0.5, ymax=3.8,
    xtick={1.0, 2.1},
    xticklabels={$x_1$, $x_2$},
    ytick={1.1, 2.794},
    yticklabels={$f(x_1)$, $f(x_2)$},
    clip=false,
    xlabel={$x$},
    ylabel={$y$},
    title style={at={(0.5,1.08)},anchor=south,color=themefg,font=\bfseries\small},
    title={严格递增：$x_1 < x_2 \implies f(x_1) < f(x_2)$},
]
  % Curve
  \addplot[themecurve, line width=2.2pt, domain=0.2:3.0, samples=150] {0.4*x^2 + 0.3*x + 0.4};

  % Test points
  % 两个标记点必须严格落在源函数上：f(1)=1.1，f(2.1)=2.794。
  \addplot[only marks, mark=*, mark size=2.8pt, fill=themecurve, draw=themefg] coordinates {(1.0, 1.1) (2.1, 2.794)};
  \draw[dotted, themegray, thick] (axis cs:1.0, 1.1) -- (axis cs:1.0, 0);
  \draw[dotted, themegray, thick] (axis cs:1.0, 1.1) -- (axis cs:0, 1.1);
  \draw[dotted, themegray, thick] (axis cs:2.1, 2.794) -- (axis cs:2.1, 0);
  \draw[dotted, themegray, thick] (axis cs:2.1, 2.794) -- (axis cs:0, 2.794);
\end{axis}

% Left bottom subcaption (Completely OUTSIDE axis, no collision with xticks)
\node[font=\small\bfseries, color=themecurve, anchor=north] at ($(axleft.south)-(0, 0.45cm)$) {严格递增 $\implies$ 单射：$x_1<x_2$ 时函数值严格分离};

% Right: Non-strictly Increasing (Platform interval)
\begin{axis}[
    at={($(axleft.east)+(2.4cm,0)$)},
    anchor=west,
    name=axright,
    width=7.5cm,
    height=7.2cm,
    axis lines=middle,
    axis line style={color=themefg, -{Stealth}},
    tick style={color=themefg},
    tick label style={color=themefg, font=\small},
    every axis label/.style={color=themefg},
    xmin=-0.5, xmax=4.0,
    ymin=-0.5, ymax=3.8,
    xtick={0.8, 1.6, 2.6},
    xticklabels={$x_1$, $x_a$, $x_b$},
    ytick={1.4, 1.8},
    yticklabels={$f(x_1)$, $C$},
    clip=false,
    xlabel={$x$},
    ylabel={$y$},
    title style={at={(0.5,1.08)},anchor=south,color=themefg,font=\bfseries\small},
    title={单调不减：允许局部平台常数区间},
]
  % Piecewise nondecreasing curve with a genuine flat plateau.
  % 第一段在 x=1.2 精确到达 1.8，保证拼接后仍是单值函数且连续不降。
  \addplot[themealert, line width=2.2pt, domain=0.2:1.2, samples=50] {x + 0.6};
  \addplot[themealert, line width=2.8pt, domain=1.2:2.8, samples=50] {1.8};
  \addplot[themealert, line width=2.2pt, domain=2.8:3.6, samples=50] {0.65*(x-2.8)^2 + 1.8};

  % Plateau points (xa, xb) with same value C
  \addplot[only marks, mark=*, mark size=2.8pt, fill=themealert, draw=themefg] coordinates {(1.6, 1.8) (2.6, 1.8)};
  \draw[dotted, themegray, thick] (axis cs:1.6, 1.8) -- (axis cs:1.6, 0);
  \draw[dotted, themegray, thick] (axis cs:2.6, 1.8) -- (axis cs:2.6, 0);
  \draw[dashed, themealert, thick] (axis cs:0, 1.8) -- (axis cs:1.2, 1.8);

  % Plateau annotation safely placed well above plateau
  \node[text=themealert, fill=themebg, inner sep=0.8pt, anchor=south] at (axis cs:2.1, 2.2) {\small 平台区：$f(x_a)=f(x_b)=C$};
\end{axis}

% Right bottom subcaption (Completely OUTSIDE axis)
\node[font=\small\bfseries, color=themealert, anchor=north] at ($(axright.south)-(0, 0.45cm)$) {若存在常数平台，则非单射；在实际值域上不存在逆映射};

\end{tikzpicture}
\end{CJK*}
\end{document}
