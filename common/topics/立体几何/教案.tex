\section{备课补充：公式来源与数学直觉}

\teachblock{线面角：高、射影、斜线}{
  线面角不是凭空间直觉猜一个角，而是把斜线投影到平面里。

  若 \(AH\perp\alpha\)，\(P,H\in\alpha\)，则 \(HP\) 是 \(AP\) 在平面 \(\alpha\) 内的射影。
  \[
    \sin\theta
    =
    \frac{AH}{AP},
    \qquad
    \cos\theta
    =
    \frac{HP}{AP}.
  \]

  课堂上可以说：高度看正弦，影子看余弦。

  \linePlaneAngleModel
}

\teachblock{向量夹角：先共起点，再数量积}{
  向量夹角公式本质上来自数量积的几何定义。
  \[
    \vec u\cdot\vec v
    =
    |\vec u|\,|\vec v|\cos\theta.
  \]

  所以：
  \[
    \cos\theta
    =
    \frac{\vec u\cdot\vec v}{|\vec u|\,|\vec v|}.
  \]

  课堂追问学生：你求的是哪两条向量的夹角？它们有没有从同一个点出发？

  \vectorAngleModel
}

\teachblock{解三角形：边角互化}{
  正弦定理负责把“边长比例”和“角的正弦”联系起来；余弦定理负责把“第三边”和“夹角”联系起来。

  遇到条件中同时有边和角，先问一句：这题是要把边化成角，还是把角化成边？
  \[
    c^2=a^2+b^2-2ab\cos C,
    \qquad
    S=\frac12ab\sin C.
  \]

  \triangleLawModel
}

\teacherproblem{示例 1}{立体几何多结论选择}

{\small \qOneStem\qOneOptions}
\begin{center}
  \qOneFigure[1.0]
\end{center}

\begin{paracol}{2}
\teachblock{标准结论}{
  本处填写答案。

  选择题教案不只写选项，要写每个关键选项成立或不成立的原因。
}

\teachblock{标准解答 / 多视角}{
  先从已知 \(PA\perp\text{平面 }\odot O\) 出发，得到 \(PA\perp AB\)、\(PA\perp AC\) 等基础垂直关系。

  选项判断要围绕定理条件：
  \[
    l\parallel \alpha
    \quad\Longleftrightarrow\quad
    \text{找平面内一条与 }l\text{ 平行的直线}
  \]
  或
  \[
    \alpha\perp\beta
    \quad\Longleftrightarrow\quad
    \text{找一个面内线垂直另一个面}.
  \]

  对学生来说，重点不是记住答案，而是能把每个空间关系翻译成“要证明什么条件”。
}

\switchcolumn
\teachblock{学生问题}{
  根据学生作答补充。

  常见问题：只看图形直觉，不检查“线在面内”“两线相交”“平行线是否确实存在”等得分条件。
}

\teachblock{课堂处理 / 追问}{
  让学生逐个选项说出判定定理，而不是直接报答案。

  追问：如果要证明一个面垂直另一个面，你要先找哪一条线？
}

\teachblock{学案思考对应}{
  对应学案示例 1 的两个自查问题；教案中必须给出回答口径。
}
\end{paracol}

\teacherproblem{示例 2}{解三角形边角互化}

{\small \qTwoStem}
\qTwoOptions

\begin{paracol}{2}
\teachblock{标准结论}{
  本处填写答案。
}

\teachblock{标准解答}{
  先把条件中的边化成角，或把角化成边。

  一种常见路线是用正弦定理：
  \[
    a=2R\sin A,
    \qquad
    b=2R\sin B,
    \qquad
    c=2R\sin C.
  \]

  代入后再用和角关系 \(A+B=\pi-C\) 化简。

  解出三角函数值后，必须结合：
  \[
    C\in(0,\pi)
  \]
  判断角是否唯一。
}

\teachblock{多视角}{
  也可以用余弦定理把 \(\cos A,\cos B\) 改写成边长表达式。

  这条路线适合训练代数化简，但课堂主线通常选更短的边角互化法。
}

\switchcolumn
\teachblock{学生问题}{
  根据学生作答补充。

  常见问题：一看到 \(\cos A,\cos B\) 就盲目套余弦定理，没有先判断正弦定理是否更顺。
}

\teachblock{课堂处理 / 追问}{
  追问：条件里有 \(a\cos B+b\cos A\)，为什么它可能和 \(c\cos C\) 合并？

  让学生说出“边角互化”的方向，再开始计算。
}

\teachblock{学案思考对应}{
  对应学案示例 2：先选方法，再检查角的范围。
}
\end{paracol}

\teacherproblem{示例 3}{平面向量夹角}

{\small \qThreeStem}
\qThreeOptions

\begin{paracol}{2}
\teachblock{标准结论}{
  本处填写答案。
}

\teachblock{标准解答}{
  先展开数量积：
  \[
    \vec a\cdot(\vec a+\vec b)
    =
    \vec a\cdot\vec a+\vec a\cdot\vec b.
  \]

  又因为 \(|\vec a|=1\)，所以：
  \[
    \vec a\cdot\vec a=|\vec a|^2=1.
  \]

  由题意可求 \(\vec a\cdot\vec b\)，再代入夹角公式：
  \[
    \cos\theta
    =
    \frac{\vec a\cdot\vec b}{|\vec a|\,|\vec b|}.
  \]

  最后由 \(\theta\in[0,\pi]\) 确定夹角。
}

\switchcolumn
\teachblock{学生问题}{
  根据学生作答补充。

  常见问题：展开数量积时漏掉 \(\vec a\cdot\vec a\)，或把 \(|\vec a|^2\) 写成 \(|\vec a|\)。
}

\teachblock{课堂处理 / 追问}{
  让学生先口述三句话：

  第一，数量积可以分配。

  第二，\(\vec a\cdot\vec a=|\vec a|^2\)。

  第三，夹角公式最后还要除以两个模长。
}

\teachblock{学案思考对应}{
  对应学案示例 3：先展开，再代夹角公式。
}
\end{paracol}

\teacherproblem{示例 4}{立体几何证明与线面角}

{\small \qFourStem}
\begin{center}
  \qFourFigure[0.95]
\end{center}

\begin{paracol}{2}
\teachblock{标准结论}{
  本处填写两个小问的结论。
}

\teachblock{标准解答}{
  第 (1) 问要写成完整证明链。

  先由已知角度得到垂直关系，再找平面内两条相交直线，使用线面垂直判定。
  \[
    \left.
    \begin{aligned}
      l&\perp a\\
      l&\perp b\\
      a\cap b&\ne\varnothing\\
      a,b&\subset\alpha
    \end{aligned}
    \right\}
    \Rightarrow
    l\perp\alpha.
  \]

  再由面面垂直判定得到结论。

  第 (2) 问先找 \(PB\) 在平面 \(ABCD\) 内的射影。

  线面角一定落在“斜线、射影、高”构成的直角三角形中，不要凭图认角。
}

\teachblock{得分公式}{
  体积公式：
  \[
    V=\frac13S_{\text{底}}h.
  \]

  线面角：
  \[
    \sin\theta=\frac{\text{点到平面的距离}}{\text{斜线长}},
    \qquad
    \cos\theta=\frac{\text{射影长}}{\text{斜线长}}.
  \]
}

\switchcolumn
\teachblock{学生问题}{
  根据学生作答补充。

  常见问题：证明题只写“显然垂直”，没有写出两条相交直线；求角时找错射影。
}

\teachblock{课堂处理}{
  让学生先不计算，只完成三件事：

  画出辅助线。

  口述要证明的判定定理。

  指出线面角所在的直角三角形。
}

\teachblock{追问}{
  为什么证明线面垂直必须强调“两条相交直线”？

  为什么线面角不能直接看原图中的某个表面角？
}

\teachblock{学案思考对应}{
  对应学案示例 4：教案中要明确回答“先找面内线垂直另一个面”和“先找射影”。
}
\end{paracol}
