\section*{一、立体几何核心知识清单与公式查漏}

\begin{taskbox}{1. 几何体结构、特征与体积公式}
\begin{itemize}[leftmargin=1.5em, itemsep=0.8em]
  \item \textbf{多面体与旋转体特征}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 棱柱/棱锥/棱台：直棱柱侧棱\blank{1.5cm}底面；正棱锥顶点投影在底面\blank{2.0cm}；棱台的所有侧棱延长线\blank{2.0cm}。
    \item 旋转体表面积：圆柱 $S_{\text{表}} = \blank{2.5cm}$；圆锥 $S_{\text{表}} = \blank{2.5cm}$；球 $S_{\text{表}} = \blank{2.0cm}$。
  \end{itemize}
  \item \textbf{体积公式与直观图}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 体积：柱体 $V = \blank{2.0cm}$；锥体 $V = \blank{2.0cm}$；台体 $V = \blank{4.5cm}$；球 $V = \blank{2.0cm}$。
    \item 斜二测画法面积关系：$S_{\text{直观图}} = \blank{1.5cm} S_{\text{原图}}$。
  \end{itemize}
  \item \textbf{正棱锥的两大计算直角三角形}：由高、侧高与\blank{2.5cm}构成；或由高、侧棱与\blank{2.5cm}构成。
\end{itemize}
\end{taskbox}

\begin{taskbox}{2. 位置关系证明与角度、距离求法}
\begin{itemize}[leftmargin=1.5em, itemsep=0.8em]
  \item \textbf{平行转化}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 线面判定（线线 $\rightarrow$ 线面）：若 $l \not\subset \alpha, m \subset \alpha$ 且 \blank{2.0cm} $\Rightarrow l \parallel \alpha$。
    \item 面面性质（面面 $\rightarrow$ 线线）：若 $\alpha \parallel \beta, \gamma \cap \alpha = m, \gamma \cap \beta = n \Rightarrow$ \blank{2.0cm}。
  \end{itemize}
  \item \textbf{垂直转化}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 线面判定：若直线 $l$ 垂直于平面内的\textbf{两条}\blank{2.0cm}直线 $\Rightarrow l \perp \alpha$。
    \item 面面性质：若 $\alpha \perp \beta, \alpha \cap \beta = m, l \subset \beta$ 且 \blank{2.0cm} $\Rightarrow l \perp \alpha$。
  \end{itemize}
  \item \textbf{三大角求法}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 异面直线夹角：平移相交，范围\blank{2.0cm}。
    \item 线面角：斜线与\blank{2.5cm}所成的角。口诀：“高度看\blank{1.5cm}，影子看\blank{1.5cm}”。
    \item 二面角：在两面内找交线的\blank{2.5cm}。或利用面积比法：$\cos\theta = \blank{2.5cm}$。
  \end{itemize}
  \item \textbf{点面距离}：首选\textbf{等体积法}（换底）：$V_{A-BCD} = V_{B-ACD} \implies h_A = \blank{4.0cm}$。
\end{itemize}
\end{taskbox}

\begin{taskbox}{3. 核心球模型与辅助线直觉心法}
\begin{itemize}[leftmargin=1.5em, itemsep=0.8em]
  \item \textbf{外接球核心模型}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 墙角模型（三棱两两垂直）：$R = \blank{3.5cm}$。
    \item 侧棱垂直底面模型：$R^2 = \blank{3.5cm}$（$r$ 为底面外心半径，$h$ 为高）。
    \item 对棱相等模型：将三棱锥嵌入\blank{2.0cm}中。
  \end{itemize}
  \item \textbf{题感联想与辅助线}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.3em]
    \item 看到中点：首选寻找或构造\blank{2.5cm}线证平行，或直角三角形斜边中线。
    \item 看到菱形/正方形：立刻联想对角线\blank{2.5cm}证垂直。
    \item 隐藏垂直来源：等腰/等边三角形的\blank{2.0cm}线；圆内直径所对的\blank{2.0cm}角。
  \end{itemize}
  \item \textbf{逆推顺证推导}：
  草稿本上“执果索因”逆推，但答题卡上\textbf{严禁}逆推，必须“执因导果”按顺序书写：已知线长关系 $\rightarrow$ 证明平面内线线垂直 $\rightarrow$ 结合已知证明线面垂直 $\rightarrow$ 结论。
\end{itemize}
\end{taskbox}
