\section{专题说明与学习目标}

\begin{decisionscan}
  本专题训练的是：会做题以后，怎样把理由、公式和结论写清楚。重点不是增加新的知识点，而是把已经会用的公式、定理和推理过程写成卷面上能得分的步骤，减少因条件缺失、范围不明、步骤跳跃造成的失分。

  本专题先看常见大题中的书写问题，再重点训练立体几何与解三角形中的条件书写、范围说明、定理引用和结论检验。
\end{decisionscan}

\subsection{学习顺序}
\begin{enumerate}[label=\arabic*. , leftmargin=2em, itemsep=0.15em, topsep=0.2em]
  \item 先完成诊断练习，找出自己在表达中的常见问题；
  \item 再学习逻辑符号、书写规范和各板块表达模板；
  \item 通过典型例题和专题训练，把规范表达落实到完整解题过程；
  \item 最后完成问题答案改写，并用表达检查清单进行复盘。
\end{enumerate}

\section{逻辑符号与等价号使用}

\subsection{常用数学逻辑符号工具箱}
\begingroup
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\textwidth}{p{2.2cm}p{2.2cm}X}
\toprule
符号 & 读法 & 必讲提醒与典型场景\\
\midrule
\(\forall\) & 对任意 & 恒成立问题。例：\(\forall x \in I, f(x) \ge 0\)。\\ \hline
\(\exists\) & 存在 & 方程有解、有交点。例：\(\exists x \in I, f(x) = 0\)。\\ \hline
\(\exists!\) & 存在唯一 & 函数零点唯一性、唯一解。在求极值点或证明零点唯一时常用。\\ \hline
\(\Rightarrow\) & 推出 & 单向推导。例：\(f'(x) > 0 \Rightarrow f(x)\) 在区间上单调递增。\\ \hline
\(\Leftrightarrow\) & 等价于 & 双向充要转化。只有双向逻辑成立时才能使用，避免在每步计算间随意使用。\\ \hline
\(\subseteq\) & 包含于 & 集合包含、充分必要条件证明。例：\(A \subseteq B\)。\\ \hline
\(\cap\) & 交集 & 同时满足。如连立条件组、分类讨论的子分支内部。\\ \hline
\(\cup\) & 并集 & 分类合并。如分类讨论最终“综上所述”时各分支并集。\\ \hline
\(\varnothing\) & 空集 & 无解、无公共点。例：解集为 \(\varnothing\)。\\ \hline
\(\because\) & 因为 & 引用前面已证条件或已知。例：\(\because a \neq 0\)。\\ \hline
\(\therefore\) & 所以 & 得到阶段性或最终结论。例：\(\therefore a > 1\)。\\ \hline
\(p \Rightarrow q\) & \(p\) 推出 \(q\) & 表示 \(p\) 是 \(q\) 的\textbf{充分条件}。\\ \hline
\(p \Leftarrow q\) & \(q\) 推出 \(p\) & 表示 \(p\) 是 \(q\) 的\textbf{必要条件}。\\
\bottomrule
\end{tabularx}
\endgroup

\begin{tcolorbox}[
  enhanced,
  colback=white,
  colframe=black,
  boxrule=0.8pt,
  arc=0pt,
  left=8pt, right=8pt, top=6pt, bottom=6pt,
  title={\small\bfseries 【提示】谨慎使用等价符号（\(\Leftrightarrow\)）}
]
  {\small
  书写解答题时，只有双向逻辑成立的步骤才可以使用 \(\Leftrightarrow\)。在很多代数变形中，\textbf{双向等价并不成立}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0.1em, topsep=0.2em]
    \item \textbf{可以写等价的场景（恒等变形）}：
      \(x + 1 > 0 \Leftrightarrow x > -1\)。
    \item \textbf{不能写等价的场景（单向推出）}：
      \(x^2 = 4 \Rightarrow x = \pm 2\)。若逆推 \(x = \pm 2 \Rightarrow x^2=4\) 成立，则此处可写等价；但对于 \(x = 2 \Rightarrow x^2=4\)，逆推并不成立（\(x^2=4\) 不能推出 \(x=2\)），不能写成等价。
    \item \textbf{平方或开方必须补条件}：
      解无理方程 \(\sqrt{x} = x - 2\)，若直接平方得到 \(x = (x-2)^2\)，必须在前面或后面补充限制条件：
      \[
        \sqrt{x} = x - 2 \Leftrightarrow \begin{cases} x - 2 \ge 0, \\ x = (x-2)^2 \end{cases}
      \]
      否则只写平方将导致产生增根且逻辑不完整。
  \end{itemize}
  }
\end{tcolorbox}

\section{卷面书写规范与写法标准}

\mindsetcard{卷面书写规范}{把思路写成得分步骤}{
  解答题按步骤给分。规范卷面应写清“必要文字 + 关键条件组 + 结论”：先说明依据，再列出条件，最后写出结论。
}

\subsection{解答题书写“六步法”}
\begin{decisionscan}
  \begin{enumerate}[label=\arabic*. , leftmargin=2em, itemsep=0.1em, topsep=0.2em]
    \item \textbf{设}：引入研究对象（如设函数、参数、线、面），建立推理起点；
    \item \textbf{限}：写明取值限制（如定义域、分母非零、斜率存在性），避免后续推理超出条件；
    \item \textbf{联}：联立方程或列出定理前置条件，确立代数与几何约束；
    \item \textbf{推}：写出由已知到结论的推理步骤，避免跨度过大，不用“显然”代替必要证明；
    \item \textbf{算}：呈现关键计算步骤与化简过渡，避免只写结果；
    \item \textbf{综}：检验边界与退化点，对分类讨论的分支进行合并收尾。
  \end{enumerate}
  \textbf{书写口诀}：先设后限，联立再推；关键计算，综上收尾。
\end{decisionscan}

\section{各板块表达规范}

\subsection{板块一：代数含参最值与分类讨论}
\begin{decisionscan}
  \textbf{动轴定区间最值模板}：\\
  1. \textbf{配方定位}：写出二次函数的配方式及对称轴 \(x = x_0\)；\\
  2. \textbf{分类分段}：根据对称轴在区间左侧、内部、右侧，明确写出“当 \dots{} 时”并分段求解：
  \begin{itemize}[leftmargin=1.5em, itemsep=0em, topsep=0.1em]
    \item 当轴在左侧时，\(f(x)_{\min} = f(\text{左端点}) = \dots\)（求得分支范围，并与分类前提求交集）；
    \item 当轴在内部时，\(f(x)_{\min} = f(\text{顶点}) = \dots\)（求得分支范围，并与分类前提求交集）；
    \item 当轴在右侧时，\(f(x)_{\min} = f(\text{右端点}) = \dots\)（求得分支范围，并与分类前提求交集）；
  \end{itemize}
  3. \textbf{综上收尾}：另起一行写“综上所述，参数的取值范围是\dots”（取各分支范围的并集）。
\end{decisionscan}

\subsection{板块二：立体几何证明条件组}
\begin{decisionscan}
  \textbf{线面垂直判定定理模板}：\\
  一定要用大括号列齐判定定理的四要素，禁止简写成单行推出：
  \[
    \text{因为 }
    \begin{cases}
      l \perp a, \\
      l \perp b, \\
      a \cap b = A, \\
      a, b \subset \text{平面 } \alpha
    \end{cases}
    \Rightarrow l \perp \text{平面 } \alpha.
  \]
  证明面面垂直时，必须明确写出线在面内条件：因为 \(l \subset \text{平面 } \beta\)，由面面垂直判定定理得平面 \(\beta \perp\) 平面 \(\alpha\)。
\end{decisionscan}

\subsection{板块三：解三角形表达要点}
\begin{decisionscan}
  \textbf{1. SSA 多解判定完整对照表}（已知对边 \(a\)、邻边 \(b\) 和夹角 \(A\)）：
  
  当 \(A\) 为锐角时（设高 \(h = b\sin A\)）：
  \begin{center}
    \begin{tabular}{|p{2.2cm}|p{3.2cm}||p{2.2cm}|p{3.2cm}|}
      \hline
      条件关系 & 三角形个数 & 条件关系 & 三角形个数 \\ \hline
      \(a < h\) & 无解 & \(h < a < b\) & 两解 \\ \hline
      \(a = h\) & 一解 (直角三角形) & \(a \ge b\) & 一解 \\ \hline
    \end{tabular}
  \end{center}
  
  当 \(A\) 为直角或钝角时：
  \begin{center}
    \begin{tabular}{|p{2.2cm}|p{3.2cm}||p{2.2cm}|p{3.2cm}|}
      \hline
      条件关系 & 三角形个数 & 条件关系 & 三角形个数 \\ \hline
      \(a > b\) & 一解 & \(a \le b\) & 无解 \\ \hline
    \end{tabular}
  \end{center}
  
  \textbf{2. 面积公式夹角检查}：使用 \(S = \frac{1}{2}bc\sin A\) 时，角 \(A\) 必须是两边 \(b, c\) 的夹角。\\
  \textbf{夹角检查方法}：面积公式先看夹角，再看对应边。\\
  \textbf{3. 边角对应检查}：正弦/余弦定理中，边角必须准确对应：\(a \leftrightarrow A, b \leftrightarrow B, c \leftrightarrow C\)。\\
  \textbf{4. 弧度与角度统一}：同一道题中尽量统一使用角度制 or 弧度制，绝不可混写。
\end{decisionscan}

\subsection{板块四：解析几何“五步式”表达模板}
\begin{decisionscan}
  \textbf{解析几何高考大题书写五步四段式}：\\
  1. \textbf{第一段：设点设线}。若斜率可能不存在，需进行分类讨论：“当直线斜率不存在时，\dots；当直线斜率存在时，设其方程为 \(y = k(x-x_0)\)，设 \(A(x_1, y_1), B(x_2, y_2)\)”。\\
  2. \textbf{第二段：联立消元}。联立直线与圆锥曲线方程，整理出消元后的一元二次方程：
  \[
    Ax^2 + Bx + C = 0.
  \]
  3. \textbf{第三段：判别式条件}。必须写出并检验：\(\Delta = B^2 - 4AC > 0\)。\\
  4. \textbf{第四段：韦达定理}。写出 \(x_1+x_2 = -\frac{B}{A}, x_1x_2 = \frac{C}{A}\)。\\
  5. \textbf{第五段：目标式代换}。列出目标关系式（如弦长、面积、向量点积等），将韦达定理整体代入化简并综合收尾。
\end{decisionscan}

\subsection{板块五：导数单调性规范链条}
\begin{decisionscan}
  \textbf{导数解题标准书写链}：\\
  \(\boxed{\text{定义域}} \rightarrow \boxed{\text{求导化简}} \rightarrow \boxed{f'(x)=0 \text{ 求根}} \rightarrow \boxed{\text{说明导号(列表)}} \rightarrow \boxed{\text{写区间}} \rightarrow \boxed{\text{写极值}}\)。\\
  \textbf{常见漏分点}：导数的正负号必须和具体区间一起写，不能只写一个不等式。\\
  \textbf{规范写法}：当 \(x \in (a, b)\) 时，\(f'(x) > 0\)，故 \(f(x)\) 在 \((a, b)\) 上单调递增。
\end{decisionscan}

\subsection{板块六：数列推理过程模板}
\begin{decisionscan}
  \textbf{1. 累加法与递推边界}：由 \(a_{n+1} - a_n = f(n)\)，必须在卷面上写明“当 \(n \ge 2\) 时”，最后回代 \(n=1\) 检验是否成立。\\
  \textbf{2. 错位相减求和}：清晰书写原式与乘公比后的式子，对齐相减，写出等比求和代入，注意尾项符号。\\
  \textbf{3. 数学归纳法三步走}：
  \begin{itemize}[leftmargin=1.5em, itemsep=0em, topsep=0.1em]
    \item 当 \(n = 1\) 时，验证命题成立；
    \item 假设当 \(n = k\) (\(k \ge 1\)) 时命题成立；
    \item 证明当 \(n = k+1\) 时命题成立。最后必须写总结性结论。
  \end{itemize}
\end{decisionscan}

\subsection{板块七：概率统计与事件定义}
\begin{decisionscan}
  \textbf{概率解答题表达模板}：\\
  1. \textbf{定义事件}：必须先明确写出“设事件 \(A\) 为\dots，事件 \(B\) 为\dots”；\\
  2. \textbf{分布说明}：在使用公式前说明随机变量服从哪一种分布（如二项分布 \(X \sim B(n,p)\)，超几何分布等）；\\
  3. \textbf{独立性声明}：在使用乘法公式 \(P(AB) = P(A)P(B)\) 前，必须在卷面上明确声明：“因为事件 \(A\) 与事件 \(B\) 相互独立，所以\dots”，否则直接相乘会被扣去步骤分。
\end{decisionscan}

  交卷前，请用 30 秒扫描解答题，检查是否有以下“隐性丢分点”：
  \begin{enumerate}[label=\checkbox , leftmargin=2em, itemsep=0.15em, topsep=0.2em]
    \item \textbf{研究对象定义}：代数题中是否写了“设 \(f(x) = \dots\)”，解析几何中是否设了点与直线？
    \item \textbf{定义域与取值限制}：对数题是否标明定义域？导数是否写了定义域？解析几何是否讨论了斜率不存在？

    \item \textbf{分类讨论收尾}：分类讨论算完后，最后是否写了“综上所述”并合并了结论？
    \item \textbf{定理前提条件组}：立体几何线面垂直证明中，大括号里是否写齐了“相交”和“线在面内”？
    \item \textbf{解三角形一角两解}：正弦定理求角时，是否漏掉了钝角解？是否用“内角和小于 \(180^\circ\)”做了存在性检验？

    \item \textbf{导数区间与符号}：求导大题里，极值、单调区间的结果是否已经和导数符号对应起来？
    \item \textbf{数列边界项与回代}：数列利用前 \(n\) 项和求通项时，是否注明了 \(n \ge 2\)，且最后进行了 \(n=1\) 检验？
    \item \textbf{概率事件与模型}：概率题是否设了“事件 \(A\) 为\dots”，使用期望前是否写了 \(X \sim B(n,p)\)，乘法前是否声明了独立？
  \end{enumerate}
