\section{教学框架、双分类体系与符号规范}

\teachblock{一、 常用数学符号及其几何含义规范}{%
向学生明确说明各数学符号的准确几何含义：
\begin{itemize}[itemsep=0.2em]
  \item \(A, B, C\)：\quad 三角形的三个内角，取值区间为 \((0,\pi)\)，满足 \(A+B+C = \pi\)；
  \item \(a, b, c\)：\quad 角 \(A, B, C\) 所对的三条对边长；
  \item \(R\)：\quad 三角形外接圆半径，代表整体大小的“外接圆尺度自由度”；
  \item \(r\)：\quad 三角形内切圆半径，代表整体大小的“内切圆尺度自由度”；
  \item \(p\)：\quad 三角形半周长，定义为 \(p = \frac{a+b+c}{2}\)；\(2p = a+b+c\) 为周长；
  \item \(m_a\)：\quad 边 \(a\) 上的中线长；\(h_a = b\sin C = c\sin B\) 为边 \(a\) 上的高线长；
  \item \(\text{DOF}\)：\quad 自由度（Degree of Freedom），指完全确定三角形几何状态所需最少独立变量数。
\end{itemize}
}

    解答“题目给了哪些条件，\textbf{我应该从哪条定理开始}”；

    解答“\textbf{为什么这些条件足以唯一确定三角形}”的几何物理本质。
\end{itemize}
}

\section{第一套分类：按已知量组合分类 (操作层入口)}

每一类已知量组合均明确【已知量是什么】与【目标要求什么】，指导起手第一步公式。

\vspace{0.5em}

\teachblock{1. SSS 型（已知三边）}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：三条边长 \(a, b, c\)；
  \item \textbf{目标要求}：三个内角 \(A, B, C\)，或面积 \(S\)，或判断三角形形状（锐/直/钝角）；
  \item \textbf{起手公式}：余弦定理求角：
    \[
    \cos A = \frac{b^2+c^2-a^2}{2bc}, \qquad S = \sqrt{p(p-a)(p-b)(p-c)} \quad (\text{海伦公式})
    \]
  \item \textbf{解题口诀}：\textit{三边求角，余弦当家}。
\end{itemize}
}

\vspace{0.3em}

\teachblock{2. SAS 型（已知两边及夹角）}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：两条边长 \(b, c\) 及其夹角 \(A\)；
  \item \textbf{目标要求}：第三条边长 \(a\)，或剩余内角 \(B, C\)；
  \item \textbf{起手公式}：先用余弦定理求第三边 \(a\)：
    \[
    a = \sqrt{b^2+c^2-2bc\cos A}
    \]
    再用正弦定理求剩余角：\(\sin B = \frac{b\sin A}{a}\)；
  \item \textbf{解题口诀}：\textit{两边夹角，余弦先上}。
\end{itemize}
}

\vspace{0.3em}

\teachblock{3. ASA / AAS 型（已知两角及一边）}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：两个内角 \(A, B\) 及任意一条边长（如 \(a\)）；
  \item \textbf{目标要求}：第三个内角 \(C\)，或剩余两边长 \(b, c\)；
  \item \textbf{起手公式}：先求第三角 \(C = \pi - A - B\)，再用正弦定理：
    \[
    b = \frac{a\sin B}{\sin A}, \qquad c = \frac{a\sin C}{\sin A}
    \]
  \item \textbf{解题口诀}：\textit{两角一边，正弦到底}。
\end{itemize}
}

\vspace{0.3em}

\teachblock{4. SSA 型（已知两边及其中一边的对角）}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：两条边长 \(a, b\) 以及边 \(a\) 的对角 \(A\)（注意：\(A\) 非 \(a,b\) 的夹角）；
  \item \textbf{目标要求}：内角 \(B\) 的大小，或第三边长 \(c\)，或讨论构成三角形解的个数（0/1/2个）；
  \item \textbf{起手公式}：用正弦定理求角 \(B\) 的正弦值：
    \[
    \sin B = \frac{b\sin A}{a}
    \]
    分情况讨论 \(B = \theta\) 或 \(B = \pi - \theta\)（须满足 \(A+B < \pi\)）；
  \item \textbf{解题口诀}：\textit{两边对角，分支讨论}。
\end{itemize}
}

\vspace{0.3em}

\teachblock{5. AAA 型（已知三个角）}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：三个内角 \(A, B, C\)（满足 \(A+B+C = \pi\)）；
  \item \textbf{目标要求}：三边长度比例 \(a : b : c\)；
  \item \textbf{起手公式}：由正弦定理得 \(a : b : c = \sin A : \sin B : \sin C\)（缺尺度量，不能求具体边长）；
  \item \textbf{解题口诀}：\textit{三角已知，只有形状}。
\end{itemize}
}

\section{第二套分类：按自由度与自然坐标分类 (结构层)}

一个三角形在几何上被\textbf{2 个形状自由度 \((A,B)\)} 和 \textbf{1 个尺度自由度 \(\lambda\)} 完全决定：
\[
\boxed{\text{三角形} = \text{形状自由度 }(A,B) + \text{尺度自由度 }\lambda}
\]

\subsection{\((A,B,R)\) 外接圆坐标系统（最自然坐标系）}

\teachblock{\((A,B,R)\) 坐标系推导}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：两个内角 \(A, B\) 与外接圆半径 \(R\)；
  \item \textbf{目标要求}：第三角 \(C\) 以及三边具体长度 \(a, b, c\)；
  \item \textbf{计算步骤与公式}：
    \[
    C = \pi - A - B
    \]
    \[
    a = 2R\sin A, \qquad b = 2R\sin B, \qquad c = 2R\sin C
    \]
\end{itemize}
}

\subsection{\((A,B,r)\) 内切圆坐标系统}

\teachblock{\((A,B,r)\) 坐标系求解步骤}{%
\begin{itemize}[itemsep=0.2em]
  \item \textbf{已知量}：两个内角 \(A, B\) 与内切圆半径 \(r\)；
  \item \textbf{目标要求}：三边具体长度 \(a, b, c\) 或外接圆半径 \(R\)；
  \item \textbf{计算步骤与公式}：
    \[
    C = \pi - A - B
    \]
    由切线段公式直接求三边：
    \[
    a = r\left(\cot\frac{B}{2} + \cot\frac{C}{2}\right), \quad
    b = r\left(\cot\frac{C}{2} + \cot\frac{A}{2}\right), \quad
    c = r\left(\cot\frac{A}{2} + \cot\frac{B}{2}\right)
    \]
    或先求外接圆半径 \(R\)：
    \[
    R = \frac{r}{4\sin\frac{A}{2}\sin\frac{B}{2}\sin\frac{C}{2}}
    \]
\end{itemize}
}

\teachblock{核心公式完整推导：\(r = 4R\sin\frac{A}{2}\sin\frac{B}{2}\sin\frac{C}{2}\)}{%
\textbf{【推导路线一：面积桥梁法】}
\begin{enumerate}[label=\arabic*., itemsep=0.3em]
  \item 联立三角形面积公式的两种表达：
    \[
    S = rp = \frac{abc}{4R} \implies r = \frac{abc}{4Rp}
    \]
  \item 由正弦定理边角互化，将分子 \(abc\) 展开：
    \[
    a = 2R\sin A = 4R\sin\frac{A}{2}\cos\frac{A}{2}
    \]
    \[
    abc = 64 R^3 \sin\frac{A}{2}\cos\frac{A}{2} \sin\frac{B}{2}\cos\frac{B}{2} \sin\frac{C}{2}\cos\frac{C}{2}
    \]
  \item 将半周长 \(p = \frac{a+b+c}{2} = R(\sin A + \sin B + \sin C)\) 展开：
    利用三角和恒等式 \(\sin A + \sin B + \sin C = 4\cos\frac{A}{2}\cos\frac{B}{2}\cos\frac{C}{2}\)，得：
    \[
    p = 4R\cos\frac{A}{2}\cos\frac{B}{2}\cos\frac{C}{2}
    \]
  \item 代入 \(r = \frac{abc}{4Rp}\) 化简：
    \[
    r = \frac{64 R^3 \sin\frac{A}{2}\cos\frac{A}{2} \sin\frac{B}{2}\cos\frac{B}{2} \sin\frac{C}{2}\cos\frac{C}{2}}{4R \cdot 4R\cos\frac{A}{2}\cos\frac{B}{2}\cos\frac{C}{2}} = 4R\sin\frac{A}{2}\sin\frac{B}{2}\sin\frac{C}{2}
    \]
    两端变形即得：
    \[
    \boxed{R = \frac{r}{4\sin\frac{A}{2}\sin\frac{B}{2}\sin\frac{C}{2}}}
    \]
\end{enumerate}

\vspace{0.3em}
\textbf{【推导路线二：切线段与半角余切法】}
\begin{enumerate}[label=\arabic*., itemsep=0.3em]
  \item 顶点 \(A\) 引出的切线段长 \(p-a = r\cot\frac{A}{2}\)，故半周长：
    \[
    p = (p-a)+(p-b)+(p-c) = r\left(\cot\frac{A}{2} + \cot\frac{B}{2} + \cot\frac{C}{2}\right)
    \]
  \item 利用恒等式 \(\cot\frac{A}{2} + \cot\frac{B}{2} + \cot\frac{C}{2} = \cot\frac{A}{2}\cot\frac{B}{2}\cot\frac{C}{2}\)，得：
    \[
    p = r \cot\frac{A}{2}\cot\frac{B}{2}\cot\frac{C}{2}
    \]
  \item 联立 \(p = 4R\cos\frac{A}{2}\cos\frac{B}{2}\cos\frac{C}{2}\)，约去余弦项即可证毕。
\end{enumerate}
}

\section{动态几何全景：一条总主线、六大母轨迹与三个综合应用}

\begin{tcolorbox}[colback=yellow!5,colframe=yellow!60!black,arc=2mm,title=\textbf{【运动学总主线法则】}]
固定底边 \(AB\)，使第三个顶点 \(C\) 在平面中运动。顶点 \(C\) 初始具有 2 个自由度 \((x,y)\)。\\
每增加 1 个独立的几何约束条件，自由度减 1：
\[
\boxed{\text{2 个自由度} - 1 \text{ 个约束} = \text{一条动点轨迹}}
\]
\textbf{解题四步法}：
1. 哪些点/边固定？ \quad 2. 哪个点在动？ \quad 3. 条件能翻译为什么关系？ \quad 4. 对应哪条母轨迹？
\end{tcolorbox}

\begin{paracol}{2}
\teachblock{标准解答与得分句}{%
\textbf{解：}
(1) 由正弦定理：
\[
\sin B = \frac{b}{2R},\quad \sin C = \frac{c}{2R}
\]
代入已知条件 \(5c\sin B = 6a\sin C\)：
\[
5c \cdot \frac{b}{2R} = 6a \cdot \frac{c}{2R} \implies 5b = 6a \implies b = \frac{6}{5}a
\]
代入齐次条件 \(b+c = 2a\)：
\[
\frac{6}{5}a + c = 2a \implies c = \frac{4}{5}a
\]
故三边长之比为：
\[
a : b : c = a : \frac{6}{5}a : \frac{4}{5}a = 5 : 6 : 4
\]
\textbf{【得分点：消元表达三边比】}

\vspace{0.3em}
(2) 由余弦定理：
\[
\cos B = \frac{a^2 + c^2 - b^2}{2ac}
\]
代入 \(b = \frac{6}{5}a, c = \frac{4}{5}a\) 得：
\[
\cos B = \frac{a^2 + \left(\frac{4}{5}a\right)^2 - \left(\frac{6}{5}a\right)^2}{2 \cdot a \cdot \left(\frac{4}{5}a\right)} = \frac{1}{8}
\]
\textbf{【得分点：余弦定理求角】}
}

\switchcolumn
\teachblock{第 25 题框架代入}{%
\textbf{处理方式}：\textcolor{red}{\textbf{精讲（25 分钟）}}

\textbf{【第 25 题本质】}：\\
第 25 题条件 \(b+c=2a\) 与 \(5c\sin B = 6a\sin C\) \textbf{完全齐次}。

无法求具体边长，本质是：\\
\textbf{利用 2 个独立形状条件，锁定相似类型（\(\text{DOF}=2\)）}。

\textbf{解题链条}：
\[
\text{边角混合} \xrightarrow{\text{正弦}} \text{纯边比}
\]
\[
\xrightarrow{b+c=2a} \text{三边比} \xrightarrow{\text{余弦}} \cos B
\]

\textbf{【课堂追问】}：\\
若条件改为 \(S = \frac{\sqrt{3}}{4}a^2\)，控制形状还是大小？\\
（答：两端均二次齐次，只控形状，解出 \(C=60^\circ\)）。
}
\end{paracol}

\teacherproblem{例题 3}{SSA 离散分支与解的个数}

{\small \qTriangleDofSsaBranchStem}

\begin{center}
\qTriangleDofSsaBranchFigure[0.95]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答与得分句}{%
\textbf{解：}
(1) 正弦定理：
\[
\sin B = \frac{b\sin A}{a} = \frac{2 \cdot \frac{\sqrt{2}}{2}}{\sqrt{3}} = \frac{\sqrt{6}}{3}
\]

(2) 几何高 \(h = b\sin A = \sqrt{2}\)。
因为：
\[
h = \sqrt{2} < a = \sqrt{3} < b = 2
\]
由 SSA 划弧交点图，圆弧与射线交于两点 \(B_1, B_2 \implies\) \textbf{2 个解}（钝角解 + 锐角解）。

(3) 余弦定理：
\[
a^2 = b^2 + c^2 - 2bc\cos A
\]
\[
3 = 4 + c^2 - 2\sqrt{2}c \implies c^2 - 2\sqrt{2}c + 1 = 0
\]
解方程得：
\[
c = \sqrt{2} \pm 1
\]
}

\switchcolumn
\teachblock{SSA 离散分支机制}{%
\textbf{处理方式}：\textcolor{red}{\textbf{精讲（25 分钟）}}

\textbf{【SSA 的本质】}：\\
反三角函数在 \((0,\pi)\) 内不能唯一确定角：
\[
\sin B = k \implies B=\theta
\]
\[
\text{或 } B = \pi-\theta
\]
产生几何分支。

\textbf{【追问】}：\\
若 \(a < h\)、\(a=h\)、\(a \ge b\) 分别有几个解？\\
（答：0 个、1 个直角、1 个解）。
}
\end{paracol}

\teacherproblem{例题 4}{几何结构与自由度传递 (中线与向量)}

{\small \qTriangleDofVectorMedianStem}

\begin{center}
\qTriangleDofVectorMedianFigure[0.95]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答与得分句}{%
\textbf{解：}
(1) 倍长中线 \(AD\) 至 \(E\) 构成平行四边形 \(ABEC\)，对角线 \(AE=2m_a=\sqrt{19}\)，\(BC=a\)。
由平行四边形恒等式：
\[
2(b^2+c^2) = a^2 + (2m_a)^2
\]
\[
2(25) = a^2 + 19 \implies a = 5
\]

(2) 余弦定理：
\[
\cos A = \frac{4^2+3^2-5^2}{2 \cdot 4 \cdot 3} = 0 \implies A = 90^\circ
\]

(3) 重心向量分解：
\[
\vec{GA} = -\frac{1}{3}(\vec{AB}+\vec{AC})
\]
\[
\vec{GB} = \frac{2}{3}\vec{AB}-\frac{1}{3}\vec{AC}
\]
因 \(A=90^\circ \implies \vec{AB} \cdot \vec{AC} = 0\)：
\[
\begin{aligned}
\vec{GA} \cdot \vec{GB} &= -\frac{2}{9}|\vec{AB}|^2 + \frac{1}{9}|\vec{AC}|^2 \\
&= -\frac{2}{9}(9) + \frac{1}{9}(16) = -\frac{2}{9}
\end{aligned}
\]
}

\switchcolumn
\teachblock{自由度分析与课后落实}{%
\textbf{处理方式}：\textcolor{red}{\textbf{精讲（20 分钟）}}

\textbf{【自由度传递】}：\\
中线长 \(m_a\) 与三边 \((a,b,c)\) 在平行四边形恒等式中完全锁定。

已知 \(b,c,m_a\) 即可锁定 \(a\)，全等确立 (\(\text{DOF}=0\))。
}
\end{paracol}

\section{五大题型压缩与方法论总图}

\begin{tcolorbox}[colback=white,colframe=gray!80,arc=2mm,title=\textbf{解三角形五大基本题型压缩}]
\begin{enumerate}[label=\arabic*., itemsep=0.3em]
  \item \textbf{第一类：确定型}（目标：唯一解三角形。SSS, SAS, ASA, AAS，判断闭合条件）；
  \item \textbf{第二类：边角翻译型}（核心动作：利用正弦/余弦定理统一成全边或全角）；
  \item \textbf{第三类：存在性与解的个数}（核心检查：两边之和大于第三边、角和小于 \(\pi\)、SSA 两支、锐钝角限制）；
  \item \textbf{第四类：范围最值型}（核心逻辑：固定若干自由度 \(\to\) 剩 1 个参数 \(\to\) 确定区间 \(\to\) 单变量最值）；
  \item \textbf{第五类：恒等式与关系证明}（目标：证明关系，路线：全边化或全角化，切忌长期边角混杂）。
\end{enumerate}
