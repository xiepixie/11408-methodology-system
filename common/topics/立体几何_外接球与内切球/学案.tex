\newcommand{\sphereformulas}{%
  \arrayrulecolor{rulecolor}
  \begin{tabularx}{\textwidth}{p{4.0cm}X}
  \toprule[1.2pt]
  核心截面公式 &
  球心、截面圆心、球面上一点构成直角三角形：
  \[
    R^2=\fblank[4cm]{r^2+d^2}.
  \]
  其中 $R$ 为球半径，$r$ 为截面圆半径，$d$ 为球心到截面的距离。\\ \hline
  正方体外接球 &
  棱长为 $a$，体对角线为球直径：
  \[
    R=\fblank[3cm]{\frac{\sqrt3}{2}a}.
  \]\\ \hline
  长方体外接球 &
  长、宽、高分别为 $a,b,c$：
  \[
    R=\fblank[4.5cm]{\frac{\sqrt{a^2+b^2+c^2}}{2}}.
  \]\\ \hline
  直棱柱外接球 &
  底面外接圆半径为 $r$，高为 $h$：
  \[
    R^2=\fblank[4.5cm]{r^2+\left(\frac h2\right)^2}.
  \]\\ \hline
  圆柱外接球 &
  底面半径为 $r$，高为 $h$：
  \[
    R^2=\fblank[4.5cm]{r^2+\left(\frac h2\right)^2}.
  \]\\ \hline
  正四面体外接球 &
  棱长为 $a$：
  \[
    R=\fblank[3cm]{\frac{\sqrt6}{4}a}.
  \]\\ \hline
  多面体内切球 &
  若多面体体积为 $V$，表面积为 $S_{\text{表}}$：
  \[
    r=\fblank[3cm]{\frac{3V}{S_{\text{表}}}}.
  \]\\
  \bottomrule[1.2pt]
  \end{tabularx}
}
