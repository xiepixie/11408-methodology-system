\newcommand{\qVectorCollinearStem}{若 \(\vec{a} = (2, 1)\) 与 \(\vec{b} = (x, 3)\) 共线，则实数 \(x\) 的值为 \hfill （\quad）}

\newcommand{\qVectorCollinearOptions}{%
  \mcoptions[4]{\(6\)}{\(\dfrac{3}{2}\)}{\(\dfrac{2}{3}\)}{\(\dfrac{1}{6}\)}%
}

\newcommand{\qVectorCollinearAnswer}{A}

\newcommand{\qVectorCollinearSolution}{%
  因为 \(\vec{a} = (2, 1)\) 与 \(\vec{b} = (x, 3)\) 共线，
  所以 \(2 \times 3 - 1 \times x = 0\)，
  解得 \(x = 6\)。
  故选 A。%
}
