\teacherproblem{5}{风筝几何与三角}
\qFiveStem
\qFiveOptions
\begin{center}
  \qFiveFigure
\end{center}

\begin{paracol}{2}
\teachblock{标准结论}{正确答案：\qFiveAnswer}
\teachblock{标准解答}{\qFiveSolution}
\teachblock{备选解法（余弦定理）}{
  设 \(AE=x\)。由 \(AC=4AE\) 得 \(CE=3x\)。\\
  在 Rt\(\triangle CED\) 中，由勾股定理得 \(DE = \sqrt{CD^2 - CE^2} = 2x\)。\\
  在 Rt\(\triangle AED\) 中，由勾股定理得 \(AD = \sqrt{5}x\)。\\
  在 \(\triangle ADC\) 中，由余弦定理：\\
  \(\cos\angle ADC = \dfrac{AD^2 + CD^2 - AC^2}{2AD\cdot CD} = \dfrac{5x^2 + 13x^2 - 16x^2}{2\sqrt{5}x\cdot\sqrt{13}x} = \dfrac{1}{\sqrt{65}}\)。\\
  故 \(\sin\angle ADC = \sqrt{1 - \cos^2\angle ADC} = \dfrac{8}{\sqrt{65}}\)。\\
  所以 \(\tan\angle ADC = \dfrac{\sin\angle ADC}{\cos\angle ADC} = 8\)。
}

\teacherproblem{11}{正方体动点轨迹多选题}
\qElevenStem
\qElevenOptions
\begin{center}
  \qElevenFigure
\end{center}

\begin{paracol}{2}
\teachblock{标准结论}{正确答案：\qElevenAnswer}
\teachblock{标准解答}{\qElevenSolution}
\teachblock{备选解法（无建系向量法）}{
  设正方体三个互相垂直的边向量为基底 \(\boldsymbol{u}=\overrightarrow{AB}, \boldsymbol{v}=\overrightarrow{BC}, \boldsymbol{w}=\overrightarrow{BB_1}\)。\\
  则 \(\overrightarrow{AP} = \boldsymbol{u} + \lambda \boldsymbol{v} + \mu \boldsymbol{w}\)。\\
  对于 B 选项：平面 \(A_1BD\) 的法向量可设为 \(\boldsymbol{n} = \boldsymbol{u} + \boldsymbol{v} + \boldsymbol{w}\)。\\
  若 \(AP \perp\) 平面 \(A_1BD\)，则 \(\overrightarrow{AP} // \boldsymbol{n}\)，这要求基底系数成比例。\\
  但当 \(\lambda = \dfrac{1}{2}\) 时，\(\overrightarrow{AP} = \boldsymbol{u} + \dfrac{1}{2}\boldsymbol{v} + \mu \boldsymbol{w}\)，其 \(u\) 和 \(v\) 分量系数之比为 1:\(\dfrac{1}{2}\)，与 \(\boldsymbol{n}\) 中的 1:1 矛盾，故 B 错误；\\
  对于 C 选项：\(\overrightarrow{A_1P} = \boldsymbol{u} + \lambda \boldsymbol{v} + (\mu - 1)\boldsymbol{w}\)。当 \(\mu = \dfrac{1}{2}\) 时，\(\overrightarrow{A_1P} = \boldsymbol{u} + \lambda \boldsymbol{v} - \dfrac{1}{2}\boldsymbol{w}\)。\\
  若 \(A_1P // AB\)，则 \(\overrightarrow{A_1P}\) 的 \(v\) 和 \(w\) 分量系数必须为 0，这要求 \(-\dfrac{1}{2} = 0\)，矛盾，故 C 正确。
}

% Q15
\teacherproblem{15}{解三角形解答题}
\qFifteenStem

\begin{paracol}{2}
\teachblock{标准解答}{\qFifteenSolution}
\teachblock{备选解法（代数边化简）}{
  由 \(\boldsymbol{m}//\boldsymbol{n}\) 得 \((2c-b)\cos A = a\cos B\)。\\
  代入余弦定理公式得：\\
  \((2c-b)\dfrac{b^2+c^2-a^2}{2bc} = a\dfrac{a^2+c^2-b^2}{2ac}\) \\
  同乘 \(2bc\) 展开得：\((2c-b)(b^2+c^2-a^2) = b(a^2+c^2-b^2)\)。\\
  展开化简：\(2c(b^2+c^2-a^2) - b(b^2+c^2-a^2) = b(a^2+c^2-b^2)\)。\\
  整理得：\(b^2+c^2-a^2-bc = 0 \implies a^2 = b^2+c^2-bc\)。\\
  结合余弦定理 \(a^2 = b^2+c^2-2bc\cos A\) 得 \(\cos A = \dfrac{1}{2}\)。\\
  因为 \(A \in (0,\pi)\)，所以 \(A = \dfrac{\pi}{3}\)。
}

% Q17
\teacherproblem{17}{直三棱柱平行证明与外接球}
\qSeventeenStem
\begin{center}
  \qSeventeenFigure
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{\qSeventeenSolution}
\teachblock{第一问备选解法（向量法）}{
  设 \(\overrightarrow{AB}=\boldsymbol{u}, \overrightarrow{AC}=\boldsymbol{v}, \overrightarrow{AA_1}=\boldsymbol{w}\)。\\
  点 \(D\) 为 \(B_1C\) 中点 \(\implies \overrightarrow{AD} = \dfrac{\overrightarrow{AB_1}+\overrightarrow{AC}}{2} = \dfrac{\boldsymbol{u}+\boldsymbol{v}+\boldsymbol{w}}{2}\)。\\
  点 \(E\) 为 \(A_1C_1\) 中点 \(\implies \overrightarrow{AE} = \dfrac{\overrightarrow{AA_1}+\overrightarrow{AC_1}}{2} = \dfrac{\boldsymbol{v}}{2}+\boldsymbol{w}\)。\\
  所以 \(\overrightarrow{DE} = \overrightarrow{AE} - \overrightarrow{AD} = \dfrac{\boldsymbol{w}-\boldsymbol{u}}{2} = \dfrac{1}{2}\overrightarrow{BA_1}\)。\\
  因为 \(BA_1 \subset\) 平面 \(AA_1B_1B\)，且 \(DE \not\subset\) 平面 \(AA_1B_1B\)，\\
  所以 \(DE //\) 平面 \(AA_1B_1B\)。
}
\switchcolumn
\teachblock{学生问题}{\qSeventeenDiagnosis}
\teachblock{课堂处理}{\qSeventeenTeachingNotes}
\teachblock{备课拓展（体积比）}{
  【备课小贴士】本题体积也可用平行六面体体积比例快速计算。设以 \(\boldsymbol{u},\boldsymbol{v},\boldsymbol{w}\) 为邻棱的平行六面体体积为 \(V_0 = \dfrac{\sqrt{3}}{2}x^3\)，直三棱柱体积为 \(\dfrac{1}{2}V_0\)，通过中点切割关系可证得三棱锥 \(B-CDE\) 的体积为平行六面体体积的 \(\dfrac{1}{24}\)，即 \(V = \dfrac{\sqrt{3}}{48}x^3\)。该视角可在拔高课中向学生拓展。
}
\end{paracol}

% Q19
\teacherproblem{19}{四棱锥二面角正弦最值}
\qNineteenStem
\begin{center}
  \qNineteenFigure
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{\qNineteenSolution}
\teachblock{第一问备选解法（几何法）}{
  过 \(P\) 作 \(PH \perp AC\) 于 \(H\)。因为平面 \(PAC \perp\) 平面 \(ABCD\)，所以 \(PH \perp\) 平面 \(ABCD\)。\\
  由三垂线定理得 \(PD \perp AB \iff HD \perp AB\)。\\
  在底面中，\(\triangle ABD\) 是边长为 \(2\sqrt{3}\) 的正三角形，其高 \(DF = 3\)（\(F\) 为 \(AB\) 中点）。\\
  因为 \(AC\) 平分 \(\angle DAB\)，所以 \(\angle BAC = 30^\circ\)。\\
  在 Rt\(\triangle AFH\) 中，\(AH = \dfrac{AF}{\cos 30^\circ} = 2\)，\(HF = AH\sin 30^\circ = 1\)。\\
  所以 \(HD = DF - HF = 3 - 1 = 2\)。\\
  在 Rt\(\triangle BFH\) 中，由勾股定理得 \(HB = \sqrt{HF^2 + FB^2} = \sqrt{1^2 + 3} = 2\)。\\
  因为 \(HB = HD = 2\)，且 \(PH \perp\) 平面 \(ABCD\)，\\
  所以 \(PB = \sqrt{PH^2 + HB^2} = \sqrt{PH^2 + HD^2} = PD\)。
}
