\section{备课指导与学情预判}

\teachblock{本讲目标}{%
\begin{itemize}[itemsep=0.3em]
  \item 理解诱导公式的几何对称本质，掌握“奇变偶不变，符号看象限”的底层逻辑。
  \item 掌握两角和差公式的“旋转坐标系法”与“向量点积法”推导，能复述推导步骤与几何直观。
  \item 建立三角公式的“衍生家谱树”，能够自主推导二倍角、半角、降幂、积化和差、和差化积公式。
  \item 掌握辅助角公式的代数与几何投影意义，能熟练确定辅助角的值。
  \item 掌握万能公式的代数推导与“直角三角形记忆机制”，深刻理解倒数平方关系在含切函数变形中的应用。
\end{itemize}
}

\textbf{覆盖:} 诱导公式的对称几何直观，两角和差公式的旋转系与点积法几何推导，二倍角、半角、降幂、积化和差、和差化积公式的代数衍生推导。新增辅助角公式的代数与投影确定，万能公式及同角倒数平方关系的代数几何分析。\\
\textbf{不覆盖:} 复杂的三角恒等求值大题变形，解三角形中的边角综合问题。这些属于后续的恒等变形与解三角形专项。
}

\section{知识引入与物理/几何背景}

\teachblock{诱导公式的物理解释：相位平移}{
  向学生解释：$\sin(\frac{\pi}{2}+\alpha) = \cos\alpha$ 实际上是波的相位移动。正弦波与余弦波在形状上完全一致，只是在时间轴上相差了 $\frac{\pi}{2}$ 的相位。
  当我们将正弦曲线向左平移 $\frac{\pi}{2}$ 单位时，它就与余弦曲线完全重合。这就是“变名（正弦变余弦）”的物理本质。而平移的方向（向左或向右）决定了平移后的曲线是否需要乘以 $-1$（即“符号看象限”）。
}

\teachblock{旋转坐标系法与旋转矩阵的联系}{
  对于基础较好的学生，可以指出：旋转坐标系法实际上是在推导二维空间中的旋转矩阵：
  \[
  \begin{pmatrix}
  \cos(\alpha+\beta) \\
  \sin(\alpha+\beta)
  \end{pmatrix}
  =
  \begin{pmatrix}
  \cos\alpha & -\sin\alpha \\
  \sin\alpha & \cos\alpha
  \end{pmatrix}
  \begin{pmatrix}
  \cos\beta \\
  \sin\beta
  \end{pmatrix}
  \]
  这说明两角和公式的底层就是“先旋转 $\alpha$，再旋转 $\beta$，等价于一次性旋转 $\alpha+\beta$”。
}

\section{课堂核心题目与讲法剖析}

\teacherproblem{1}{引入: 诱导公式的几何直观与对称性}

{\small \qTrigInductionStem}

\begin{center}
  \qTrigInductionFigure[1.0]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{%
  在单位圆中，设角 $\alpha$ 的终边与单位圆交于 $P_1(x, y)$。
  则由三角函数的定义：
  \[ \cos\alpha = x, \quad \sin\alpha = y \]
  
  角 $\frac{\pi}{2}+\alpha$ 是将射线 $OP_1$ 逆时针旋转 $90^\circ$ 得到的射线 $OP_2$。
  如图，作 $P_1, P_2$ 在坐标轴上的投影，构成直角三角形。
  由于 $OP_1 = OP_2 = 1$，且 $\angle P_2 O y = \angle P_1 O x = \alpha$，
  易证 $\triangle P_2 O M_2 \cong \triangle P_1 O N_1$（其中 $M_2$ 为 $P_2$ 在 $x$ 轴上的投影，$N_1$ 为 $P_1$ 在 $y$ 轴上的投影）。
  
  因此，在长度上：
  $P_2$ 的横坐标绝对值等于 $P_1$ 的纵坐标绝对值，即 $|x'| = |y|$；
  $P_2$ 的纵坐标绝对值等于 $P_1$ 的横坐标绝对值，即 $|y'| = |x|$。
  
  考虑到 $P_2$ 位于第二象限，其横坐标为负，纵坐标为正，故有：
  \[ x' = -y, \quad y' = x \]
  
  根据三角函数定义，得：
  \[ \cos\left(\frac{\pi}{2}+\alpha\right) = x' = -y = -\sin\alpha \]
  \[ \sin\left(\frac{\pi}{2}+\alpha\right) = y' = x = \cos\alpha \]
  
  这便从几何全等上证明了诱导公式。
}

\teachblock{板书建议}{%
  重点在黑板上用彩色粉笔画出 $\triangle P_1 O M_1$ 与 $\triangle P_2 O M_2$ 两个直角三角形，标出全等角，直观演示“横纵坐标长度交换”的过程。
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{错因表现：} 学生最容易犯的错误是“符号看象限”看错了目标。
  例如在化简 $\cos(\frac{\pi}{2}+\alpha)$ 时，有些学生会推导为 $\cos(\frac{\pi}{2}+\alpha) = \sin\alpha$，其理由是：$\alpha$ 视为锐角，$\frac{\pi}{2}+\alpha$ 在第二象限，而正弦（第二象限）为正，所以结果是正号。
  
  这犯了“符号看变好后的函数（正弦）”的低级错误，正确的应该是看原函数（余弦在第二象限为负）。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  让学生在单位圆的第一、二象限用手比划横坐标的值。
  提问：“看图，$\cos(\frac{\pi}{2}+\alpha)$ 代表的是 $P_2$ 的横坐标。横坐标在第二象限是正还是负？”引导学生脱口而出“负”。
  
  再问：“它在长度上跟 $P_1$ 的哪个轴坐标一样长？”学生回答“$P_1$ 的纵坐标，也就是 $\sin\alpha$”。从而牢固建立 $-\sin\alpha$ 的认知。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “如果 $\alpha$ 本身不是锐角，而是 $120^\circ$，这个几何推导全等三角形的关系还成立吗？为什么我们仍能直接用公式？”
  （引导：通过诱导公式的代数推广，任意角均可通过相差整数倍的周期转为第一象限，所以公式在复数域/任意角均恒成立，视 $\alpha$ 为锐角只是一种代数符号提取手段。）
}
\end{paracol}

\teacherproblem{2}{引入: 两角和差公式的旋转系与点积法推导}

{\small \qTrigAdditionDerivationStem}

\begin{center}
  \qTrigRotationFigure[0.9] \qquad \qTrigDotProductFigure[0.9]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{%
  \textbf{(1) 旋转直角坐标系法：}
  
  在平面直角坐标系中，设 $\mathbf{e}_1 = (\cos\alpha, \sin\alpha)$。
  将其逆时针旋转 $90^\circ$，得到与其垂直的单位向量：
  \[ \mathbf{e}_2 = (-\sin\alpha, \cos\alpha) \]
  
  设单位向量 $\overrightarrow{OQ}$ 与 $\mathbf{e}_1$ 的夹角为 $\beta$，则在以 $(\mathbf{e}_1, \mathbf{e}_2)$ 为正交基底的新直角坐标系中，$\overrightarrow{OQ}$ 的坐标为 $(\cos\beta, \sin\beta)$。
  根据向量的线性组合，有：
  \[ \overrightarrow{OQ} = \cos\beta \mathbf{e}_1 + \sin\beta \mathbf{e}_2 \]
  
  代入基底在原坐标系下的坐标：
  \[ 
  \begin{aligned}
  \overrightarrow{OQ} &= \cos\beta (\cos\alpha, \sin\alpha) + \sin\beta (-\sin\alpha, \cos\alpha) \\
  &= (\cos\alpha\cos\beta - \sin\alpha\sin\beta, \sin\alpha\cos\beta + \cos\alpha\sin\beta)
  \end{aligned}
  \]
  
  另一方面，$\overrightarrow{OQ}$ 与正 $x$ 轴的夹角为 $\alpha+\beta$。由三角函数定义，其在标准坐标系下的坐标又应是：
  \[ \overrightarrow{OQ} = (\cos(\alpha+\beta), \sin(\alpha+\beta)) \]
  
  对比横、纵坐标分量，即得：
  \[ \cos(\alpha+\beta) = \cos\alpha\cos\beta - \sin\alpha\sin\beta \]
  \[ \sin(\alpha+\beta) = \sin\alpha\cos\beta + \cos\alpha\sin\beta \]
  
  \textbf{(2) 向量点积法：}
  
  设单位向量 $\mathbf{u}=(\cos\alpha, \sin\alpha)$，$\mathbf{v}=(\cos\beta, \sin\beta)$。它们的夹角为 $\alpha-\beta$。
  由向量点积定义，$\mathbf{u} \cdot \mathbf{v} = |\mathbf{u}||\mathbf{v}|\cos(\alpha-\beta)$。
  因为 $\mathbf{u}, \mathbf{v}$ 是单位向量，所以：
  \[ \mathbf{u} \cdot \mathbf{v} = \cos(\alpha-\beta) \]
  
  根据点积的坐标运算法则：
  \[ \mathbf{u} \cdot \mathbf{v} = \cos\alpha\cos\beta + \sin\alpha\sin\beta \]
  
  两式相等，立刻得到：
  \[ \cos(\alpha-\beta) = \cos\alpha\cos\beta + \sin\alpha\sin\beta \]
  
  在此公式中用 $-\beta$ 替换 $\beta$，利用余弦偶函数和正弦奇函数：
  \[ \cos(\alpha+\beta) = \cos\alpha\cos(-\beta) + \sin\alpha\sin(-\beta) = \cos\alpha\cos\beta - \sin\alpha\sin\beta \]
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{1. 新基底向量分解困难：} 学生在理解“旋转坐标系法”时，最难理解的是新基底向量的分解公式 $\overrightarrow{OQ} = \cos\beta \mathbf{e}_1 + \sin\beta \mathbf{e}_2$。他们习惯了在静止坐标系下工作，不容易接受把向量看作新基底的线性组合。
  
  \textbf{2. 和差公式转换脱节：} 点积法极度简洁，但学生往往不知道如何从 $\cos(\alpha-\beta)$ 出发推出正弦和差公式，卡在诱导公式转化的代数步骤上。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  \textbf{1. 类比标准系：} 给学生做类比：在原坐标系中，任何单位向量都可以写成 $\cos\theta \mathbf{i} + \sin\theta \mathbf{j}$。如果我们把整个坐标轴旋转了 $\alpha$，现在的基底是 $\mathbf{e}_1, \mathbf{e}_2$，那么一个在新轴里偏转了 $\beta$ 的向量，自然就是 $\cos\beta \mathbf{e}_1 + \sin\beta \mathbf{e}_2$。
  
  \textbf{2. 诱导公式衔接：} 引导学生做诱导公式替换，利用 $\sin\theta = \cos(\frac{\pi}{2}-\theta)$，令 $\theta = \alpha+\beta$：
  \[ \sin(\alpha+\beta) = \cos\left[\left(\frac{\pi}{2}-\alpha\right)-\beta\right] \]
  用已证的差角公式展开，即可得到正弦和角公式。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “在旋转直角系法中，为什么垂直基底 $\mathbf{e}_2$ 的坐标是 $(-\sin\alpha, \cos\alpha)$？你可以用诱导公式 $\cos(\alpha+90^\circ)$ 和 $\sin(\alpha+90^\circ)$ 来说明吗？”
  2. “点积法得到的是 $\cos(\alpha-\beta)$，如果我们的图画成 $\alpha < \beta$，点积的结果和公式会有改变吗？”
  （引导：点积是标量，夹角是绝对值 $|\alpha-\beta|$，且余弦是偶函数，所以公式完全对称不变。）
}
\end{paracol}

\teacherproblem{3}{巩固: 二倍角、半角与降幂公式的代数衍生推导}

{\small \qTrigDoubleHalfStem}

\begin{center}
  \qTrigFormulaTreeFigure[0.9]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{%
  \textbf{(1) 二倍角公式：}
  在和角公式中令 $\beta = \alpha$：
  \[ \sin 2\alpha = \sin(\alpha+\alpha) = \sin\alpha\cos\alpha + \cos\alpha\sin\alpha = 2\sin\alpha\cos\alpha \]
  \[ \cos 2\alpha = \cos(\alpha+\alpha) = \cos\alpha\cos\alpha - \sin\alpha\sin\alpha = \cos^2\alpha - \sin^2\alpha \]
  
  利用 $\sin^2\alpha + \cos^2\alpha = 1$ 进行代换：
  第一种形式（仅含余弦）：代入 $\sin^2\alpha = 1-\cos^2\alpha$：
  \[ \cos 2\alpha = \cos^2\alpha - (1-\cos^2\alpha) = 2\cos^2\alpha - 1 \]
  第二种形式（仅含正弦）：代入 $\cos^2\alpha = 1-\sin^2\alpha$：
  \[ \cos 2\alpha = (1-\sin^2\alpha) - \sin^2\alpha = 1 - 2\sin^2\alpha \]
  
  正切二倍角：由 $\tan(\alpha+\beta) = \frac{\tan\alpha+\tan\beta}{1-\tan\alpha\tan\beta}$，令 $\beta = \alpha$ 得：
  \[ \tan 2\alpha = \frac{2\tan\alpha}{1-\tan^2\alpha} \]
  
  \textbf{(2) 降幂公式：}
  从余弦二倍角公式的后两种形式移项：
  由 $\cos 2\alpha = 1 - 2\sin^2\alpha \Longrightarrow \sin^2\alpha = \frac{1-\cos 2\alpha}{2}$
  by $\cos 2\alpha = 2\cos^2\alpha - 1 \Longrightarrow \cos^2\alpha = \frac{1+\cos 2\alpha}{2}$
  
  \textbf{(3) 半角公式：}
  在降幂公式中令 $\alpha = \frac{\theta}{2}$：
  \[ \sin^2\frac{\theta}{2} = \frac{1-\cos\theta}{2} \Longrightarrow \sin\frac{\theta}{2} = \pm\sqrt{\frac{1-\cos\theta}{2}} \]
  \[ \cos^2\frac{\theta}{2} = \frac{1+\cos\theta}{2} \Longrightarrow \cos\frac{\theta}{2} = \pm\sqrt{\frac{1+\cos\theta}{2}} \]
  其中正负号由半角 $\frac{\theta}{2}$ 所在的象限决定。
  
  正切半角公式的推导（无根号版）：
  \[ \tan\frac{\theta}{2} = \frac{\sin\frac{\theta}{2}}{\cos\frac{\theta}{2}} = \frac{2\sin^2\frac{\theta}{2}}{2\sin\frac{\theta}{2}\cos\frac{\theta}{2}} = \frac{1-\cos\theta}{\sin\theta} \]
  \[ \tan\frac{\theta}{2} = \frac{2\sin\frac{\theta}{2}\cos\frac{\theta}{2}}{2\cos^2\frac{\theta}{2}} = \frac{\sin\theta}{1+\cos\theta} \]
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{1. 忽视半角象限约束：} 学生在解半角公式时，极易忽视“正负号由半角决定”这一约束。常犯错误：根据 $\theta$ 在第二象限判定正余弦，从而在开方求 $\sin\frac{\theta}{2}$ 时机械套用正号，而没有做多值象限讨论。
  
  \textbf{2. 倍角形式选择困难：} 在化简或证明题中，学生往往在面对 $\cos 2x$ 时不知道该选三种形式中的哪一种，缺乏“消常数1”的解题直觉。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  \textbf{1. 专项训练：} 若 $\theta$ 是第二象限角，求 $\frac{\theta}{2}$ 所在象限。板书范围推导如下：
  \[ \theta \in \left(\frac{\pi}{2}+2k\pi, \pi+2k\pi\right) \]
  两边除以 2：
  \[ \frac{\theta}{2} \in \left(\frac{\pi}{4}+k\pi, \frac{\pi}{2}+k\pi\right) \]
  当 $k$ 为偶数时在第一象限（正）；当 $k$ 为奇数时在第三象限（负）。以此强调必须做象限分类讨论。
  
  \textbf{2. 引导“消 1”口诀：} 
  $\bullet$ 看到 $1 - \cos 2x$，用 $\cos 2x = 1 - 2\sin^2x$，使其变为 $2\sin^2x$（常数 1 消失）；
  $\bullet$ 看到 $1 + \cos 2x$，用 $\cos 2x = 2\cos^2x - 1$，使其变为 $2\cos^2x$（常数 1 消失）。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “已知 $\cos\alpha = -\frac{7}{25}$，且 $\alpha \in (\pi, 2\pi)$。求 $\tan\frac{\alpha}{2}$。你有几种解法？哪一种不需要讨论开方正负号？”
  （引导：使用无根号版公式 $\tan\frac{\alpha}{2} = \frac{1-\cos\alpha}{\sin\alpha}$，只需先求出 $\sin\alpha$ 的值，因为 $\alpha$ 在第三/四象限，可以直接计算，避开半角正负讨论，速度最快。）
}
\end{paracol}

\teacherproblem{4}{提高: 积化和差与和差化积公式的对称换元推导}

{\small \qTrigProductSumStem}

\begin{paracol}{2}
\teachblock{标准解答}{%
  \textbf{(1) 积化和差公式的推导：}
  写出正弦的和角与差角公式：
  \[ \sin(\alpha+\beta) = \sin\alpha\cos\beta + \cos\alpha\sin\beta \quad \text{--- (1)} \]
  \[ \sin(\alpha-\beta) = \sin\alpha\cos\beta - \cos\alpha\sin\beta \quad \text{--- (2)} \]
  
  (1) + (2) 得：
  \[ \sin(\alpha+\beta) + \sin(\alpha-\beta) = 2\sin\alpha\cos\beta \]
  由此可得：
  \[ \sin\alpha\cos\beta = \frac{1}{2}[\sin(\alpha+\beta) + \sin(\alpha-\beta)] \]
  
  (1) - (2) 得：
  \[ \cos\alpha\sin\beta = \frac{1}{2}[\sin(\alpha+\beta) - \sin(\alpha-\beta)] \]
  
  同理，将余弦的和角与差角公式相加与相减：
  \[ \cos(\alpha+\beta) = \cos\alpha\cos\beta - \sin\alpha\sin\beta \quad \text{--- (3)} \]
  \[ \cos(\alpha-\beta) = \cos\alpha\cos\beta + \sin\alpha\sin\beta \quad \text{--- (4)} \]
  
  (3) + (4) 得：
  \[ \cos\alpha\cos\beta = \frac{1}{2}[\cos(\alpha+\beta) + \cos(\alpha-\beta)] \]
  
  (4) - (3) 得（注意减法顺序）：
  \[ \sin\alpha\sin\beta = \frac{1}{2}[\cos(\alpha-\beta) - \cos(\alpha+\beta)] \]
  
  \textbf{(2) 和差化积公式的推导：}
  利用换元法，设 $x = \alpha+\beta$，$y = \alpha-\beta$。
  解这个方程组，得到：
  \[ \alpha = \frac{x+y}{2}, \quad \beta = \frac{x-y}{2} \]
  
  将其代入积化和差公式：
  代入 $\sin\alpha\cos\beta = \frac{1}{2}[\sin(\alpha+\beta) + \sin(\alpha-\beta)]$，得：
  \[ \sin\frac{x+y}{2}\cos\frac{x-y}{2} = \frac{1}{2}[\sin x + \sin y] \]
  两边乘以 2：
  \[ \sin x + \sin y = 2\sin\frac{x+y}{2}\cos\frac{x-y}{2} \]
  
  同理可得：
  \[ \sin x - \sin y = 2\cos\frac{x+y}{2}\sin\frac{x-y}{2} \]
  \[ \cos x + \cos y = 2\cos\frac{x+y}{2}\cos\frac{x-y}{2} \]
  \[ \cos x - \cos y = -2\sin\frac{x+y}{2}\sin\frac{x-y}{2} \]
}

\teachblock{“中心-偏差”换元直观：}{%
  教给学生这种数学结构美：任意两个不同的角 $x$ 和 $y$，它们的“中心”是平均角 $\frac{x+y}{2}$，“偏移量”是半差角 $\frac{x-y}{2}$。
  这就像数轴上的两个点，可以通过中点和半长来表示。代数公式的背后其实是对称性。
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{1. 漏记余弦差积负号：} 在推导或背诵和差化积公式时，学生极易漏掉 $\cos x - \cos y$ 公式前面的负号。
  
  \textbf{2. 积化和差项序颠倒：} 在积化和差中，$\sin\alpha\sin\beta$ 后面是 $\cos(\alpha-\beta) - \cos(\alpha+\beta)$。学生容易把两项的顺序记反。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  \textbf{1. 负号检验法：} 让学生代入特殊值自检。
  比如令 $x = \frac{\pi}{2}$，$y = 0$。
  左边 $= \cos\frac{\pi}{2} - \cos 0 = 0 - 1 = -1$。
  右边 $= -2\sin\frac{\pi/2+0}{2}\sin\frac{\pi/2-0}{2} = -2\sin\frac{\pi}{4}\sin\frac{\pi}{4} = -2 \cdot \frac{\sqrt{2}}{2} \cdot \frac{\sqrt{2}}{2} = -1$。
  
  告诉学生，如果不加负号，右边算出来就是正数，和左边矛盾。考场上如果不确定负号，花 5 秒钟带特值（如 $\frac{\pi}{2}$ 和 $0$）就能绝对保证正确性。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “已知 $\sin\theta\sin\phi = \frac{1}{4}$，且 $\cos(\theta-\phi) = \frac{3}{4}$，你能求出 $\cos(\theta+\phi)$ 的值吗？它使用了哪个公式？”
  （引导：直接利用积化和差公式中 $\sin\alpha\sin\beta$ 的那一项展开即可，不需要求出 $\theta$ 和 $\phi$。）
}
\end{paracol}

\teacherproblem{5}{巩固: 辅助角公式的几何投影与代数推导}

{\small \qTrigAuxiliaryStem}

\begin{center}
  \qTrigAuxiliaryFigure[1.0]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{%
  对于表达式 $a\sin x + b\cos x$（其中 $a,b$ 不同时为 0）。
  
  第一步：提取公因式 $\sqrt{a^2+b^2}$，变形为：
  \[ a\sin x + b\cos x = \sqrt{a^2+b^2} \left[ \frac{a}{\sqrt{a^2+b^2}}\sin x + \frac{b}{\sqrt{a^2+b^2}}\cos x \right] \]
  
  第二步：由于系数满足平方和为 1 的关系：
  \[ \left( \frac{a}{\sqrt{a^2+b^2}} \right)^2 + \left( \frac{b}{\sqrt{a^2+b^2}} \right)^2 = 1 \]
  根据三角函数定义，必存在唯一的角 $\varphi \in [0, 2\pi)$ 使得：
  \[ \cos\varphi = \frac{a}{\sqrt{a^2+b^2}}, \quad \sin\varphi = \frac{b}{\sqrt{a^2+b^2}} \]
  且此时辅助角满足 $\tan\varphi = \frac{b}{a}$。
  
  第三步：代入上式，应用两角和的正弦公式：
  \[ 
  \begin{aligned}
  a\sin x + b\cos x &= \sqrt{a^2+b^2} [ \cos\varphi\sin x + \sin\varphi\cos x ] \\
  &= \sqrt{a^2+b^2} \sin(x+\varphi)
  \end{aligned}
  \]
  
  这便完成了代数推导。
}

\teachblock{“看点定角”几何投影法}{%
  教师在板书时，必须画出直角坐标系中的点 $P(a, b)$，标出终边向量 $OP$ 及其倾斜角 $\varphi$。
  向学生说明：点 $P$ 所在的象限，就是辅助角 $\varphi$ 所在的象限。这提供了唯一确定 $\varphi$ 值的最直观几何途径。
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{象限误判是第一死穴。}
  学生在求解例如 $-\sqrt{3}\sin x + \cos x$ 的化简时，常常机械地套用代数公式得出：
  \[ \tan\varphi = \frac{b}{a} = \frac{1}{-\sqrt{3}} \Longrightarrow \varphi = -\frac{\pi}{6} \]
  他们未能结合点 $(a,b) = (-\sqrt{3}, 1)$ 所在的第二象限进行判定，从而误将辅助角当成了 $-\frac{\pi}{6}$。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  让学生把正弦的系数 $a$ 和余弦的系数 $b$ 看作点 $P(a, b)$。
  
  提问：“对于 $-\sqrt{3}\sin x + \cos x$，点 $P(-\sqrt{3}, 1)$ 落在第几象限？”学生回答：“第二象限。”
  
  再问：“第二象限里，切值为 $-\frac{\sqrt{3}}{3}$ 的角是多少度？”学生能立刻推导出 $\frac{5\pi}{6}$（即 $150^\circ$）。从而避开代数公式的象限死角。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “若式子为 $a\sin x - b\cos x$，我们在找点 $P$ 时，对应的横纵坐标是什么？”
  （引导：应该将 $-b$ 看作整体系数，即对应的点为 $P(a, -b)$，随后通过点 $P$ 的象限来确定辅助角的值。）
}
\end{paracol}

\teacherproblem{6}{提高: 万能公式与同角三角函数的基本关系}

{\small \qTrigUniversalStem}

\begin{center}
  \qTrigUniversalFigure[1.0]
\end{center}

\begin{paracol}{2}
\teachblock{标准解答}{%
  \textbf{(1) 万能公式代数推导：}
  设 $t = \tan\frac{\alpha}{2}$。由倍角与平方关系：
  \[ 
  \sin\alpha = 2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2} = \frac{2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2}}{\sin^2\frac{\alpha}{2} + \cos^2\frac{\alpha}{2}}
  \]
  由于 $\cos\frac{\alpha}{2} \ne 0$（即 $\alpha \ne \pi+2k\pi$），分子分母同除以 $\cos^2\frac{\alpha}{2}$：
  \[ \sin\alpha = \frac{2\tan\frac{\alpha}{2}}{1+\tan^2\frac{\alpha}{2}} = \frac{2t}{1+t^2} \]
  
  对于余弦公式：
  \[
  \cos\alpha = \cos^2\frac{\alpha}{2} - \sin^2\frac{\alpha}{2} = \frac{\cos^2\frac{\alpha}{2} - \sin^2\frac{\alpha}{2}}{\sin^2\frac{\alpha}{2} + \cos^2\frac{\alpha}{2}}
  \]
  分子分母同除以 $\cos^2\frac{\alpha}{2}$：
  \[ \cos\alpha = \frac{1-\tan^2\frac{\alpha}{2}}{1+\tan^2\frac{\alpha}{2}} = \frac{1-t^2}{1+t^2} \]
  
  \textbf{(2) 万能直角三角形记忆机制：}
  如图所示，构造一个直角边为 $2t$ 和 $1-t^2$ 的直角三角形，由勾股定理其斜边长为 $1+t^2$。在该三角形中：
  夹角为 $\alpha$ 时，直接由定义读出其正弦为对边比斜边即 $\frac{2t}{1+t^2}$，余弦为邻边比斜边即 $\frac{1-t^2}{1+t^2}$，正切为对边比邻边即 $\frac{2t}{1-t^2}$。
  
  \textbf{(3) 同角倒数平方关系式：}
  由 $\sin^2\theta + \cos^2\theta = 1$，两边同除以 $\cos^2\theta$：
  \[ \frac{\sin^2\theta}{\cos^2\theta} + 1 = \frac{1}{\cos^2\theta} \Longrightarrow \frac{1}{\cos^2\theta} = 1 + \tan^2\theta \]
  因为在导数中 $(\tan\theta)' = \frac{1}{\cos^2\theta}$，所以在化简导数大题中，这可以将含切的导数式子统一表达为 $\tan\theta$ 的多项式。
}

\switchcolumn
\teachblock{学生问题分析}{%
  【学生问题】\\
  \textbf{1. 忽视公式有意义的自变量前提限制。}
  学生在进行万能置换时，极易忽视当 $\frac{\alpha}{2} = \frac{\pi}{2} + k\pi \Rightarrow \alpha = \pi + 2k\pi$ 时，$\tan\frac{\alpha}{2}$ 无意义。
  
  如果在方程求解中未进行该边界情况的代入检验，会导致漏掉 $\alpha = \pi$ 的解。
  
  \textbf{2. 无法联想到平方倒数关系。}
  在求导中，很多学生计算 $\tan x - x$ 的单调性时，卡在导数 $y' = \frac{1}{\cos^2x} - 1$ 的化简上，未能运用倒数平方关系化为 $\tan^2x$。
}

\teachblock{课堂处理}{%
  【课堂处理】\\
  1. 强调万能公式在微积分里被称为“万能置换”的深意，它能把一切非有理的三角函数积分转化为有理分式代数。
  
  2. 引导学生解题规范：使用万能置换时，必须在步骤第一行写出：*“当 $\alpha \ne \pi+2k\pi$ 时，设 $t = \tan\frac{\alpha}{2}$ ；当 $\alpha = \pi+2k\pi$ 时，单独代入原式检验。”* 以此保证不漏解。
}

\teachblock{课堂追问}{%
  【追问】\\
  1. “求函数 $y = \tan x - x$ 在 $(-\frac{\pi}{2}, \frac{\pi}{2})$ 上的单调性，请写出求导及化简的完整过程。”
}
\end{paracol}

\section{学生常见问题汇总与教学策略}

\begin{enumerate}[label=\arabic*. , leftmargin=2em, itemsep=0.4em]
  \item \textbf{问题：觉得公式太多，记不住，或者记了就忘。}
  \\ \textbf{策略：} 上课时不要让学生死记硬背公式，而是带领他们“徒手画树”。每次上课或复习前，在白纸上顺着“和差公式 $\rightarrow$ 二倍角 $\rightarrow$ 半角/降幂”以及“和差公式 $\rightarrow$ 积化和差 $\rightarrow$ 和差化积”这两条线梳理一遍。只要记住了源头两角和差，其他所有公式都应该具备在 10 秒内推导出来的能力。
  \item \textbf{问题：“符号看象限”误判。}
  \\ \textbf{策略：} 重申法则，必须看“原函数”在象限内的正负，且必须将公式中的 $\alpha$ 视为“锐角”来判断位置。通过诱导公式的几何全等图在脑海中建立直观映射。
  \item \textbf{问题：半角公式开方漏掉负号或多解讨论。}
  \\ \textbf{策略：} 给定半角求值题，必须先写出“因为 $\alpha \in \dots$ 所以 $\frac{\alpha}{2} \in \dots$”这句得分句，以此强制逼迫学生判断象限符号。
  \item \textbf{问题：辅助角公式中对负系数 $a<0$ 时的象限卡点。}
  \\ \textbf{策略：} 坚决贯彻“看点定角”口诀，将系数提取后在纸上勾画平面点坐标，直接读出对应象限的终边角。
\end{enumerate}
