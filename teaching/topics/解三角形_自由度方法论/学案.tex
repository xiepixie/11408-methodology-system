\section{本讲目标与符号说明}

\begin{tcolorbox}[colback=white,colframe=gray!80,arc=2mm,title=extbf{符号说明与几何含义规范}]
\begin{itemize}[itemsep=0.2em]
  \item \(A, B, C\)：三角形的三个内角（满足 \(A+B+C = \pi\)）；
  \item \(a, b, c\)：内角 \(A, B, C\) 所对的三条对边长；
  \item \(R\)：外接圆半径（尺度自由度）；\quad \(r\)：内切圆半径（尺度自由度）；
  \item \(p = \frac{a+b+c}{2}\)：半周长；\quad \(2p = a+b+c\)：周长；\quad \(m_a\)：边 \(a\) 上的中线长；
  \item \(\text{DOF}\)：自由度（Degree of Freedom，决定几何状态所需独立变量数）。
\end{itemize}
\end{tcolorbox}

\vspace{0.5em}

\begin{itemize}[itemsep=0.3em]
  \item extbf{第一套分类（操作层）}：看清已知量（SSS, SAS, ASA/AAS, SSA）与求解目标，迅速选对起手工具。
  \item extbf{第二套分类（结构层）}：理解 \((A,B,R)\) 与 \((A,B,r)\) 两套自然坐标（已知角与半径求边）。

\end{itemize}

\section{双分类体系对比与决策 SOP}

\begin{tcolorbox}[colback=white,colframe=gray!80,arc=2mm,title=extbf{第一套分类：已知量组合与起手工具 (操作层)}]
\begin{itemize}[itemsep=0.4em]
  \item extbf{SSS（已知三边 \(a,b,c\)）} \(\longrightarrow\) 求解内角或面积，用 \blank{4cm} 定理；
  \item extbf{SAS（已知两边 \(b,c\) 及其夹角 \(A\)）} \(\longrightarrow\) 求解第三边 \(a\)，用 \blank{4cm} 定理；
  \item extbf{ASA / AAS（已知两角 \(A,B\) 及一边 \(a\)）} \(\longrightarrow\) 求解第三角 \(C=\pi-A-B\) 及剩余边，用 \blank{3cm} 定理；
  \item extbf{SSA（已知两边 \(a,b\) 及对角 \(A\)）} \(\longrightarrow\) 求解角 \(B\)，用正弦定理求 \(\sin B = \frac{b\sin A}{a}\) 并做 \blank{3cm} 讨论。
\end{itemize}
\end{tcolorbox}

\vspace{1.0em}

\begin{tcolorbox}[colback=white,colframe=gray!80,arc=2mm,title=extbf{第二套分类：自然坐标系 (结构层)}]
\begin{itemize}[itemsep=0.4em]
  \item extbf{\((A,B,R)\) 坐标系}：已知量为两角 \(A,B\) 与外接圆半径 \(R\)，求解目标为三边，求法：
    \[
    a = \blank{2.5cm},\qquad b = \blank{2.5cm},\qquad c = \blank{2.5cm}
    \]
  \item extbf{\((A,B,r)\) 坐标系}：已知量为两角 \(A,B\) 与内切圆半径 \(r\)，求解目标为三边，由切线段公式：
    \[
    a = r\left(\cot\frac{B}{2} + \cot\frac{C}{2}\right)
    \]
\end{itemize}
\end{tcolorbox}
