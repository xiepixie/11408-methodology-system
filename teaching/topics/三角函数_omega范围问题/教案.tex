\section{教学目标与重难点}
\textbf{【教学目标】}
\begin{enumerate}[label=\arabic*. , leftmargin=2em, itemsep=0.1em, topsep=0.2em]
  \item 深入理解正余弦型函数的相位、角频率、最小正周期之间的数理与几何关联.
  \item 熟练运用“统一变量法”，将自变量 $x$ 的区间通过换元 $\theta = \omega x + \varphi$ 转化为相位区间 $[\omega a + \varphi, \omega b + \varphi]$，以此作为解题的唯一入口.
  \item 能够处理“区间内恰有 $n$ 个特征点”（零点、对称轴、极值点）的边界问题，掌握“第 $n$ 个必入、第 $n+1$ 个必出”的临界分析法，并能精确区分开闭区间端点等号.
  \item 学会利用对称轴与对称中心的横向距离规律，求解离散的 $\omega$ 候选序列，并通过回代验证法对单调性、极值条件进行筛选.
  \item 掌握“滑动区间模型”的分析方法，理解初相位 $\varphi$ 任意可变时，区间长度与特征点分布间距的几何模型.
\end{enumerate}

\vspace{0.8em}
\textbf{【教学重难点】}
\begin{description}[leftmargin=2em, itemsep=0.1em, topsep=0.2em]
  \item[重点] 统一变量换元步骤的规范书写与相位区间长度 $\Delta\theta = \omega(b-a)$ 的物理意义；特征点计数的双向不等式临界限制.
  \item[难点] 滑动区间模型中区间长度与交替出现特征点间距的不等式转化；离散 $\omega$ 候选值回代单调性与极值的精确检验.
\end{description}

\section{核心思维：统一变量换元法}

对于形如
$$f(x) = A\sin(\omega x+\varphi) + b \quad \text{或} \quad f(x) = A\cos(\omega x+\varphi) + b,$$
我们应先令
$$\theta = \omega x + \varphi.$$
这里的 $\theta$ 称为\textbf{相位（phase）}，$\omega>0$ 称为\textbf{角频率（angular frequency）}，其最小正周期（fundamental period）为
$$T = \frac{2\pi}{\omega}.$$
若自变量定义域限定为 $x \in [a, b]$，则相位的取值区间为：
$$\theta \in [\omega a + \varphi, \omega b + \varphi],$$
相位区间的物理长度为：
$$\Delta\theta = \omega(b-a).$$

\teachblock{几何直观}{
  三角函数中的 $\omega$ 范围问题，本质上是在研究一个\textbf{左端点固定（或不固定）、长度随 $\omega$ 伸缩变动}的相位区间：
  \begin{itemize}[itemsep=0.15em, topsep=0.2em]
    \item 能够包含多少个零点、对称轴或极值点（特征点计数模型）；
    \item 能否落在某一段单调区间内（单调区间包含模型）；
    \item 能否避开所有零点（无零点避让模型）；
    \item 能否完整落在函数值非负或非正的区间内（恒正恒负模型）.
  \end{itemize}
}

\subsection{正弦函数的常用特征相位}
对于 $y = \sin\theta$，常用特征相位的对应条件为：

\arrayrulecolor{rulecolor}
\begin{tabularx}{\textwidth}{p{3.5cm}X}
\toprule[1.2pt]
\textbf{性质} & \textbf{相位条件 ($k\in\mathbb{Z}$)} \\ \midrule
零点、对称中心 & $\theta = k\pi$ \\ \midrule
对称轴 & $\theta = \frac{\pi}{2} + k\pi$ \\ \midrule
最大值点 & $\theta = \frac{\pi}{2} + 2k\pi$ \\ \midrule
最小值点 & $\theta = \frac{3\pi}{2} + 2k\pi$ \\ \midrule
单调递增区间 & $\theta \in \left[-\frac{\pi}{2} + 2k\pi, \frac{\pi}{2} + 2k\pi\right]$ \\ \midrule
单调递减区间 & $\theta \in \left[\frac{\pi}{2} + 2k\pi, \frac{3\pi}{2} + 2k\pi\right]$ \\ 
\bottomrule[1.2pt]
\end{tabularx}

\subsection{余弦函数的常用特征相位}
对于 $y = \cos\theta$，常用特征相位的对应条件为：

\begin{tabularx}{\textwidth}{p{3.5cm}X}
\toprule[1.2pt]
\textbf{性质} & \textbf{相位条件 ($k\in\mathbb{Z}$)} \\ \midrule
对称轴 & $\theta = k\pi$ \\ \midrule
零点、对称中心 & $\theta = \frac{\pi}{2} + k\pi$ \\ \midrule
单调递增区间 & $\theta \in [-\pi + 2k\pi, 2k\pi]$ \\ \midrule
单调递减区间 & $\theta \in [2k\pi, \pi + 2k\pi]$ \\ \midrule
$\cos\theta \ge 0$ (非负值) & $\theta \in \left[-\frac{\pi}{2} + 2k\pi, \frac{\pi}{2} + 2k\pi\right]$ \\
\bottomrule[1.2pt]
\end{tabularx}

\teacherproblem{7}{对称中心与对称轴的距离法}
\teachvalue{本题引入“距离分析法”. 通过对称中心与对称轴在相位轴上的奇数个四分之一周期距离关系，先求出离散的候选 $\omega$，再利用单调递减区间长度进行范围限制与回代排除，是非常经典的综合压轴题.}

{\small \qTrigOmegaSevenStem}
\qTrigOmegaSevenOptions

\begin{paracol}{2}
\teachblock{标准解答}{
  \qTrigOmegaSevenSolution
}

\teachblock{通用思路与应用}{
  正弦函数的对称中心和对称轴在相位轴上相差奇数个四分之一周期，因此横向坐标的距离必然满足：
  $$d = \frac{2n+1}{4}T = \frac{(2n+1)\pi}{2\omega}.$$
  当题目中同时给出对称中心和对称轴的横坐标时，优先使用这一距离关系，可以直接解出一组离散的 $\omega$ 候选表达式，大幅提高解题效率.
  
  需要注意的是，距离关系只保证“对称中心和对称轴存在”，并不保证单调方向（增减性）正确. 因此求得的离散候选值必须代回验证.
}

\teacherproblem{10}{初相位可变时的滑动区间模型}
\teachvalue{本题为本专题的终极难点之一. 引入“滑动区间模型”，当 $\varphi$ 任意可选时，区间起点不固定，但区间长度 $2\omega\pi$ 是固定的. 我们通过平移该区间，寻找恰能覆盖 2 个零点时区间长度的临界范围.}

{\small \qTrigOmegaTenStem}

\begin{paracol}{2}
\teachblock{标准解答}{
  \qTrigOmegaTenSolution
}

\teachblock{通用思路与应用}{
  若初相位 $\varphi$ 未知且允许自由选择，问题不再是相位区间的绝对起点，而是：
  \begin{enumerate}[itemsep=0.1em, topsep=0.1em]

  \begin{itemize}[itemsep=0.15em, topsep=0.2em]
    \item “存在 $\varphi$，使区间内恰有若干零点”；
    \item “存在初相位，使函数在区间内恒正”；
    \item “存在平移量，使区间内恰有若干极值点”.
  \end{itemize}

\teachvalue{本题作为本专题的总结压轴题，结合了正弦函数的图像读图（纵截距、升降趋势）、对称中心以及单调区间限制. 考查学生利用导数符号（或切线趋势）精准排除多余初相位、并运用对称轴与单调性求解最大 $\omega$ 的综合能力.}

{\small \qTrigOmegaTwelveStem}
\qTrigOmegaTwelveFigure[0.9]

\begin{paracol}{2}
\teachblock{标准解答}{
  \qTrigOmegaTwelveSolution
}

\teachblock{通用思路与应用}{
  仅通过纵轴交点（如 $\sin\varphi = 1/2$）往往会产生两个对称的初相位. 此时必须结合曲线经过 $y$ 轴时的升降方向：
  $$f'(0) = A\omega\cos\varphi.$$
  \begin{itemize}[itemsep=0.15em, topsep=0.2em]
    \item 图像在 $x=0$ 处上升 $\implies f'(0) > 0 \implies \cos\varphi > 0$；
    \item 图像在 $x=0$ 处下降 $\implies f'(0) < 0 \implies \cos\varphi < 0$.
  \end{itemize}
  这种结合切线方向确定初相位象限的方法，在高考读图填空题中是极为核心的高效技巧.
}

\section{本讲教学总结与板书提炼}

\begin{theorembox}{三角函数 $\omega$ 问题的统一法则}
  \begin{enumerate}[label=\arabic*. , leftmargin=2em, itemsep=0.5em, topsep=0.3em]
    \item \textbf{统一入口}：做题的第一步永远是令 $\theta = \omega x + \varphi$，将自变量 $x$ 的区间转化为相位区间. 所有的单调性、对称性、零点以及恒成立都是在相位轴 $\theta$ 上完成的.
    \item \textbf{特征点计数原则}：解决“恰有 $n$ 个特征点”的问题，绝不能仅看区间长度. 必须精确列出双向不等式组，限制“第 $n$ 个必入、第 $n+1$ 个必出”. 同时，必须严密核对开闭区间端点处是否能够取得等号.
    \item \textbf{无零点与单调恒成立的超集转化}：
      \begin{itemize}[itemsep=0.2em]
        \item “无零点”转化为相位区间是相邻两个零点闭区间 $[k\pi, (k+1)\pi]$ 的子集；
        \item “单调递增/递减/恒正”转化为相位区间是标准对应相位的子集.
      \end{itemize}
    \item \textbf{离散频率的距离法与估算筛选}：
      \begin{itemize}[itemsep=0.2em]
        \item 同时给出对称轴和对称中心横坐标时，直接利用四分之一周期的奇数倍距离关系：$d = \frac{2n+1}{4}T$ 快速生成离散 $\omega$ 候选集.
        \item 区间单调性给出的 $\omega \le \frac{\pi}{\Delta x}$ 通常只作为必要条件限制范围. 算出的候选值必须由大到小回代，检验区间内是否跨越了极值相位以及单调方向是否符合.
      \end{itemize}
    \item \textbf{滑动区间模型}：当初相位 $\varphi$ 任意可变时，区间起点是浮动的. 核心思维是考察相位区间的长度 $L = \omega(b-a)$，将其与特征点点列之间的最小、最大间距进行对比，通过平移实现精确覆盖.
  \end{enumerate}
\end{theorembox}
