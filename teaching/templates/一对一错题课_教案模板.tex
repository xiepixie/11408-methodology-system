\section{备课知识库与几何直观模型}

\teachblock{1. 线面角：“高、射影、斜线”模型}{
  线面角不是凭空间直觉猜角，而是把斜线投影到平面内。
  若 \(AH\perp\alpha\)，\(P,H\in\alpha\)，则 \(HP\) 是 \(AP\) 在平面 \(\alpha\) 内的射影，\(\angle APH=\theta\) 为线面角。
  \[
    \sin\theta = \frac{AH}{AP}\quad \text{(高度看正弦)},
    \qquad
    \cos\theta = \frac{HP}{AP}\quad \text{(影子看余弦)}.
  \]
  \linePlaneAngleModel
}

\teachblock{2. 线面垂直与面面垂直判定标准得分链}{
  得分点在于必须写全四个前置条件，缺一不可：
  \[
    \left.
    \begin{aligned}
      l &\perp a \\
      l &\perp b \\
      a &\cap b = P \\
      a, b &\subset \alpha
    \end{aligned}
    \right\} \implies l \perp \alpha,
    \qquad\qquad
    \left.
    \begin{aligned}
      l &\subset \beta \\
      l &\perp \alpha
    \end{aligned}
    \right\} \implies \beta \perp \alpha.
  \]
}

\teachblock{3. 解三角形边角互化口诀}{
  遇到同时包含边和角的表达式，先引导学生做出决策：“化边为角”还是“化角为边”？
  \[
    a = 2R\sin A,\quad b = 2R\sin B,\quad c = 2R\sin C;
    \qquad
    \cos C = \frac{a^2+b^2-c^2}{2ab}.
  \]
  \triangleLawModel
}

{\small
\begin{teachcolumns}
  \teachblock{标准解答（可板书得分版）}{
    (1) 证明：在底面中，由已知可得 \(AB \perp BC\)。\par
    又 \(PA \perp \text{平面 } ABCD\)，\(BC \subset \text{平面 } ABCD\)，\par
    所以 \(PA \perp BC\)。\par
    在平面 \(PAB\) 中，因为：
    \[
      \left.
      \begin{aligned}
        BC &\perp AB \\
        BC &\perp PA \\
        AB \cap PA &= A \\
        AB, PA &\subset \text{平面 } PAB
      \end{aligned}
      \right\} \implies BC \perp \text{平面 } PAB.
    \]
    又 \(BC \subset \text{平面 } PBC\)，\par
    所以 \(\text{平面 } PBC \perp \text{平面 } PAB\)。\\[0.2em]
    (2) 解：由 (1) 知 \(PA \perp \text{平面 } ABCD\)，故 \(PB\) 在底面射影为 \(AB\)。\par
    因此，\(\angle PBA\) 即为斜线 \(PB\) 与底面所成的角 \(\theta\)。\par
    在 \(\text{Rt}\triangle PAB\) 中，\(\tan\theta = \frac{PA}{AB}\)，由此求得 \(\theta = 45^\circ\)。
  }

  \teachblock{得分公式与关键句}{
    得分点：(1) \(PA\perp BC\) 理由；(2) 4 个判定条件齐全；(3) 射影定义与所求角认定。
  }

  \switchcolumn

  \teachblock{课前证据}{
    上次作业中证明题漏写 \(AB \cap PA = A\)；求角时把 \(\angle PBC\) 误当成线面角。
  }

  \checkpoint
    {不看提示，提问：“要求 \(PB\) 与底面的夹角，第一步先找什么？”}
    {能回答“找 \(P\) 在底面的垂足和射影”，让学生独立书写，教师减少讲解。}
    {若直接认角，先引导回到线面角定义图；若知道射影但不会证明，补线面垂直判定。}

  \teachblock{追问梯度}{
    \textbf{一级：}你需要先确定哪个点在平面上的投影？\par
    \textbf{二级：}怎样证明 \(BC\) 真的垂直于平面 \(PAB\)？相交条件写了吗？\par
    \textbf{三级：}向量法中，法向量与方向向量余弦值和线面角正弦值是什么关系？
  }

  \teachblock{学案思考对应}{
    对应学案第 1 题自查项：检查证明链中是否有相交符号 \(\cap\)。
  }
\end{teachcolumns}
